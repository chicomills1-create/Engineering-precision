     1	import { spawn, type ChildProcess } from "node:child_process";
     2	import { logger } from "./logger";
     3	
     4	const PHOENIX_OFFSET_MS = 7 * 60 * 60_000;
     5	const RECOVERY_GRACE_MINUTES = 30;
     6	
     7	export type DailyRunnerDeadline = {
     8	  slot: string;
     9	  deadline: Date;
    10	};
    11	
    12	export function getNextPhoenixDailyRunnerDeadline(
    13	  now: Date,
    14	  lastStartedSlot?: string,
    15	): DailyRunnerDeadline {
    16	  const phoenixNow = new Date(now.getTime() - PHOENIX_OFFSET_MS);
    17	  const year = phoenixNow.getUTCFullYear();
    18	  const month = phoenixNow.getUTCMonth();
    19	  const day = phoenixNow.getUTCDate();
    20	  const minutes = phoenixNow.getUTCHours() * 60 + phoenixNow.getUTCMinutes();
    21	  const primary = 20 * 60;
    22	  const recovery = primary + 30;
    23	  const recoveryGraceEnd = recovery + RECOVERY_GRACE_MINUTES;
    24	
    25	  let targetDay = day;
    26	  let targetMinutes: number;
    27	  if (minutes < primary) {
    28	    targetMinutes = primary;
    29	  } else if (minutes < recovery) {
    30	    targetMinutes = primary;
    31	  } else if (minutes < recoveryGraceEnd) {
    32	    targetMinutes = recovery;
    33	  } else {
    34	    targetDay += 1;
    35	    targetMinutes = primary;
    36	  }
    37	
    38	  const targetPhoenix = Date.UTC(
    39	    year,
    40	    month,
    41	    targetDay,
    42	    Math.floor(targetMinutes / 60),
    43	    targetMinutes % 60,
    44	  );
    45	  const deadline = new Date(targetPhoenix + PHOENIX_OFFSET_MS);
    46	  const labelDate = new Date(targetPhoenix);
    47	  const date = labelDate.toISOString().slice(0, 10);
    48	  const time = targetMinutes === primary ? "20:00" : "20:30";
    49	  const candidate = {
    50	    slot: `${date}T${time}`,
    51	    deadline: deadline.getTime() <= now.getTime() ? now : deadline,
    52	  };
    53	  if (candidate.slot !== lastStartedSlot) return candidate;
    54	
    55	  if (candidate.slot.endsWith("T20:00")) {
    56	    return {
    57	      slot: `${date}T20:30`,
    58	      deadline: new Date(Date.UTC(year, month, targetDay, 20, 30) + PHOENIX_OFFSET_MS),
    59	    };
    60	  }
    61	  const nextDayPhoenix = Date.UTC(year, month, targetDay + 1, 20, 0);
    62	  const nextDate = new Date(nextDayPhoenix).toISOString().slice(0, 10);
    63	  return {
    64	    slot: `${nextDate}T20:00`,
    65	    deadline: new Date(nextDayPhoenix + PHOENIX_OFFSET_MS),
    66	  };
    67	}

    69	function spawnDailyOutreachProcess(): ChildProcess {
    70	  return spawn(
    71	    process.execPath,
    72	    ["--enable-source-maps", "artifacts/api-server/dist/outreachDailyRunner.mjs"],
    73	    {
    74	      cwd: process.cwd(),
    75	      env: process.env,
    76	      stdio: "inherit",
    77	    },
    78	  );
    79	}

    81	export function startDailyOutreachProcessScheduler(): () => Promise<void> {
    82	  if (process.env.NODE_ENV !== "production") return async () => {};
    83	  let timer: NodeJS.Timeout | undefined;
    84	  let child: ChildProcess | undefined;
    85	  let stopping = false;
    86	  let lastStartedSlot: string | undefined;
    87	
    88	  const scheduleNext = () => {
    89	    if (stopping || child) return;
    90	    const next = getNextPhoenixDailyRunnerDeadline(new Date(), lastStartedSlot);
    91	    const delay = Math.max(0, next.deadline.getTime() - Date.now());
    92	    timer = setTimeout(() => {
    93	      timer = undefined;
    94	      if (stopping) return;
    95	      logger.info({ slot: next.slot }, "Starting durable daily outreach process");
    96	      child = spawnDailyOutreachProcess();
    97	      let failedToSpawn = false;
    98	      child.once("spawn", () => {
    99	        lastStartedSlot = next.slot;
   100	      });
   101	      child.once("error", (error) => {
   102	        failedToSpawn = true;
   103	        logger.error({ err: error, slot: next.slot }, "Durable daily outreach process failed to start");
   104	      });
   105	      child.once("close", (code, signal) => {
   106	        if (code === 0) {
   107	          logger.info({ slot: next.slot }, "Durable daily outreach process completed");
   108	        } else {
   109	          logger.error(
   110	            { code, signal, slot: next.slot },
   111	            "Durable daily outreach process failed",
   112	          );
   113	        }
   114	        child = undefined;
   115	        if (failedToSpawn && !stopping) {
   116	          timer = setTimeout(scheduleNext, 60_000);
   117	          timer.unref();
   118	        } else {
   119	          scheduleNext();
   120	        }
   121	      });
   122	    }, delay);
   123	    timer.unref();
   124	  };
   125	
   126	  const stop = async () => {
   127	    stopping = true;
   128	    if (timer) clearTimeout(timer);
   129	    if (child && child.exitCode === null) {
   130	      const activeChild = child;
   131	      activeChild.kill("SIGTERM");
   132	      await Promise.race([
   133	        new Promise<void>((resolve) => activeChild.once("close", () => resolve())),
   134	        new Promise<void>((resolve) => setTimeout(resolve, 4_000)),
   135	      ]);
   136	    }
   137	  };
   138	
   139	  scheduleNext();
   140	  logger.info("Daily outreach process scheduler enabled for 20:00 and 20:30 Phoenix");
   141	  return stop;
   142	}
