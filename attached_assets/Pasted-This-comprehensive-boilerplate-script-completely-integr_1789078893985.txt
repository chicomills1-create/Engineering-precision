main_py = """import re
from fastapi import FastAPI, HTTPException, Request
from fastapi.responses import HTMLResponse
from fastapi.templating import Jinja2Templates

app = FastAPI()
templates = Jinja2Templates(directory="templates")

DISCIPLINES = {
    "mep": "MEP Engineering",
    "structural": "Structural Engineering",
    "civil": "Civil Engineering"
}

# Regional compliance variables to satisfy Search Engine Helpful Content Guidelines
REGIONAL_DATA = {
    "CA": {"compliance": "Seismic safety design structures, California Title 24 energy budget calculations, and HCAI/OSHPD regulatory compliance.", "climate": "High-risk seismic shifting zones"},
    "FL": {"compliance": "High-Velocity Hurricane Zone (HVHV) wind load structural engineering and ASCE 7 flood design compliance.", "climate": "Coastal hurricane wind corridors"},
    "TX": {"compliance": "Expansive clay soil structural foundation stabilization, TCEQ stormwater retention planning, and IBC windstorm protocols.", "climate": "Varying severe weather terrain and shifting soil layers"},
    "NY": {"compliance": "NYCDOB modern building code tracking, high-density structural layout load parameters, and NYSERDA green performance mandates.", "climate": "Dense urban freeze-thaw seasonal cycles"},
    "IL": {"compliance": "Chicago Building Code structural adaptation models, heavy regional snow load calculation matrices, and micro-climate wind stress modeling.", "climate": "Extreme sub-zero atmospheric and seasonal wind patterns"}
}

DEFAULT_REGIONAL = {"compliance": "International Building Code (IBC) structural specifications, local regional safety configurations, and classic ASHRAE performance baselines.", "climate": "Standard localized environmental conditions"}

def clean_slug_text(text: str) -> str:
    return " ".join([word.capitalize() for word in text.split("-")])

@app.get("/{discipline}-engineering-{city}-{state}", response_class=HTMLResponse)
async def render_local_seo_page(request: Request, discipline: str, city: str, state: str):
    if discipline not in DISCIPLINES:
        raise HTTPException(status_code=404, detail="Discipline not supported")
    
    state_code = state.upper()
    if len(state_code) != 2 or not state_code.isalpha():
        raise HTTPException(status_code=404, detail="Invalid location data string format")

    formatted_discipline = DISCIPLINES[discipline]
    formatted_city = clean_slug_text(city)
    
    # Dynamically inject structural variations so search engines recognize unique topical authority per market
    region_info = REGIONAL_DATA.get(state_code, DEFAULT_REGIONAL)
    
    page_data = {
        "meta_title": f"{formatted_discipline} Firm in {formatted_city}, {state_code} | Apex Grid",
        "meta_desc": f"Top-rated commercial {formatted_discipline.lower()} consultants serving {formatted_city}, {state_code}. Expert engineers handling local municipal code, layout compliance blueprints, and infrastructural development plans.",
        "h1": f"{formatted_discipline} Services in {formatted_city}, {state_code}",
        "city": formatted_city,
        "state": state_code,
        "discipline": formatted_discipline,
        "compliance_text": region_info["compliance"],
        "climate_text": region_info["climate"]
    }

    return templates.TemplateResponse("pseo_template.html", {"request": request, "page": page_data})
"""

with open("main.py", "w", encoding="utf-8") as f:
    f.write(main_py)

# 4. Programmatic XML Sitemap Generation Core Script
sitemap_gen_py = """import xml.etree.ElementTree as ET
from xml.dom import minidom

# Sample mapping structure. Instruct Replit Agent to load an complete US database list here 
# or read from a raw text/csv asset containing all 19,000 city names.
SAMPLE_GEO_DATA = [
    ("new-york", "ny"), ("los-angeles", "ca"), ("chicago", "il"), ("houston", "tx"), ("phoenix", "az"),
    ("philadelphia", "pa"), ("san-antonio", "tx"), ("san-diego", "ca"), ("dallas", "tx"), ("san-jose", "ca"),
    ("miami", "fl"), ("orlando", "fl"), ("tampa", "fl"), ("austin", "tx"), ("atlanta", "ga")
]

BASE_URL = "https://apexgrideng.com"
DISCIPLINES = ["mep", "structural", "civil"]

def build_segmented_sitemap(discipline):
    root = ET.Element("urlset", xmlns="http://sitemaps.org")
    
    # In production, loop across your entire 19,000 global target records array
    for city, state in SAMPLE_GEO_DATA:
        url_elem = ET.SubElement(root, "url")
        loc_elem = ET.SubElement(url_elem, "loc")
        loc_elem.text = f"{BASE_URL}/{discipline}-engineering-{city}-{state}"
        
        changefreq = ET.SubElement(url_elem, "changefreq")
        changefreq.text = "monthly"
        
        priority = ET.SubElement(url_elem, "priority")
        priority.text = "0.6"
        
    xml_str = ET.tostring(root, encoding='utf-8')
    parsed_str = minidom.parseString(xml_str).toprettyxml(indent="  ")
    
    filename = f"static/sitemaps/sitemap-{discipline}-cities.xml"
    with open(filename, "w", encoding="utf-8") as f:
        f.write(parsed_str)
    print(f"Successfully compiled {filename}")

if __name__ == '__main__':
    print("Initiating programmatic segmentation build sequences...")
    for disc in DISCIPLINES:
        build_segmented_sitemap(disc)
    print("Sitemap pipeline sequences completed successfully.")
"""

with open("generate_sitemaps.py", "w", encoding="utf-8") as f:
    f.write(sitemap_gen_py)
