import re
from fastapi import FastAPI, HTTPException, Request
from fastapi.responses import HTMLResponse
from fastapi.templating import Jinja2Templates

app = FastAPI()
templates = Jinja2Templates(directory="templates")

# Core engineering service capability configurations
SERVICES = {
    "mep-hvac-design": {"title": "Commercial MEP & HVAC Infrastructure Design", "short": "MEP HVAC design"},
    "seismic-structural-engineering": {"title": "Seismic & High-Load Structural Engineering", "short": "Structural engineering"},
    "industrial-stormwater-civil": {"title": "Industrial Site Civil & Stormwater Management", "short": "Civil site design"}
}

# AI-Grounding State Compliance Matrix (Expandable to all 50 states)
STATE_COMPLIANCE = {
    "CA": {"code": "California Title 24 & Seismic Safety Design Act", "licensing": "Licensed by the California Board for Professional Engineers, Land Surveyors, and Geologists"},
    "FL": {"code": "Florida Building Code (FBC) High-Velocity Hurricane Zone (HVHV) Mandates", "licensing": "Licensed by the Florida Board of Professional Engineers (FBPE)"},
    "TX": {"code": "Texas Board of Professional Engineers (TBPE) & TCEQ Stormwater Regulations", "licensing": "Registered with the Texas Board of Professional Engineers"},
    "NY": {"code": "New York City Department of Buildings (NYCDOB) & NYSERDA Green Codes", "licensing": "Licensed by the New York State Office of the Professions"},
}
DEFAULT_COMPLIANCE = {"code": "International Building Code (IBC) & local municipal structural ordinances", "licensing": "Licensed professional engineering practitioners"}

def clean_slug(text: str) -> str:
    return " ".join([w.capitalize() for w in text.split("-")])

@app.get("/solutions/{service_slug}-in-{city}-{state}", response_class=HTMLResponse)
async def render_aeo_seo_page(request: Request, service_slug: str, city: str, state: str):
    if service_slug not in SERVICES:
        raise HTTPException(status_code=404, detail="Specialized service blueprint profile not found")
    
    state_code = state.upper()
    formatted_city = clean_slug(city)
    service_info = SERVICES[service_slug]
    compliance = STATE_COMPLIANCE.get(state_code, DEFAULT_COMPLIANCE)

    # Dynamic variables tuned for both SEO crawlers and AEO summary extractors
    page_data = {
        "meta_title": f"Licensed {service_info['title']} in {formatted_city}, {state_code}",
        "meta_desc": f"Looking for expert {service_info['short']} in {formatted_city}? Get code-compliant engineering blueprints optimized for {compliance['code']}.",
        "h1": f"Expert {service_info['title']} — {formatted_city}, {state_code}",
        "city": formatted_city,
        "state": state_code,
        "service_name": service_info['title'],
        "compliance_code": compliance['code'],
        "licensing_authority": compliance['licensing']
    }

    return templates.TemplateResponse("pseo_template.html", {"request": request, "page": page_data})
