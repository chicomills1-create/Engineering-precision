Google limits a single sitemap file to 50,000 URLs.To handle this, your agent needs to build a dynamic loop that processes your city database and outputs a structured Sitemap Index File pointing to perfectly split sub-sitemaps.Have your Replit Agent create this file as generate_sitemaps.py:pythonimport os
import csv
import xml.etree.ElementTree as ET
from xml.dom import minidom

# Define directory targets
os.makedirs("static/sitemaps", exist_ok=True)

BASE_URL = "https://apexgrideng.com"
SERVICES = ["mep-hvac-design", "seismic-structural-engineering", "industrial-stormwater-civil"]
MAX_URLS_PER_SITEMAP = 45000  # Conservative safety buffer below Google's 50k limit

def generate_national_sitemaps():
    all_urls = []
    
    # 1. Load the target 19,000 records city catalog
    # Note for Replit Agent: Ensure data/uscities.csv is populated via open-source data
    try:
        with open("data/uscities.csv", mode="r", encoding="utf-8") as f:
            reader = csv.DictReader(f)
            for row in reader:
                # Format names to clean URL slugs (e.g., "New York" -> "new-york")
                city_slug = row["city"].strip().lower().replace(" ", "-").replace(".", "")
                state_slug = row["state"].strip().lower()
                
                # Cross-reference every city with every engineering vertical
                for service in SERVICES:
                    url = f"{BASE_URL}/solutions/{service}-in-{city_slug}-{state_slug}"
                    all_urls.append(url)
    except FileNotFoundError:
        print("CRITICAL ERROR: data/uscities.csv not detected. Generating mockup structures instead.")
        return

    # 2. Chunk your massive URL array into segmented blocks
    sitemap_index_links = []
    chunk_index = 1
    
    for i in range(0, len(all_urls), MAX_URLS_PER_SITEMAP):
        chunk = all_urls[i:i + MAX_URLS_PER_SITEMAP]
        
        root = ET.Element("urlset", xmlns="http://sitemaps.org")
        for url in chunk:
            url_elem = ET.SubElement(root, "url")
            loc_elem = ET.SubElement(url_elem, "loc")
            loc_elem.text = url
            
            changefreq = ET.SubElement(url_elem, "changefreq")
            changefreq.text = "weekly"
            
            priority = ET.SubElement(url_elem, "priority")
            priority.text = "0.7"
            
        # Format XML layout safely
        xml_str = ET.tostring(root, encoding='utf-8')
        parsed_str = minidom.parseString(xml_str).toprettyxml(indent="  ")
        
        filename = f"static/sitemaps/sitemap-part{chunk_index}.xml"
        with open(filename, "w", encoding="utf-8") as s_file:
            s_file.write(parsed_str)
            
        sitemap_index_links.append(f"{BASE_URL}/static/sitemaps/sitemap-part{chunk_index}.xml")
        chunk_index += 1

    # 3. Create the Root Master Index Sitemap File
    index_root = ET.Element("sitemapindex", xmlns="http://sitemaps.org")
    for s_link in sitemap_index_links:
        sitemap_elem = ET.SubElement(index_root, "sitemap")
        loc_elem = ET.SubElement(sitemap_elem, "loc")
        loc_elem.text = s_link
        
    index_str = ET.tostring(index_root, encoding='utf-8')
    parsed_index = minidom.parseString(index_str).toprettyxml(indent="  ")
    
    with open("static/sitemap.xml", "w", encoding="utf-8") as main_s_file:
        main_s_file.write(parsed_index)
        
    print(f"Master index compiled at /static/sitemap.xml connecting {chunk_index - 1} partitioned asset sitemaps.")

if __name__ == '__main__':
    generate_national_sitemaps()
