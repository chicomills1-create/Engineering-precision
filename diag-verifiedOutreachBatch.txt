    95	export async function preflightVerifiedOutreachBatch(options: {
    96	  label: string;
    97	  target: number;
    98	  contacts: readonly VerifiedBatchContactValidationInput[];
    99	  enforceTarget?: boolean;
   100	  snapshot?: {
   101	    activeProspects: Array<{
   102	      companyName: string;
   103	      contactEmail: string | null;
   104	      website: string | null;
   105	      dedupeKey?: string | null;
   106	      contactStatus?: string | null;
   107	    }>;
   108	    suppressedEmails: string[];
   109	  };
   110	}): Promise<OutreachPreflightReport> {
   111	  assertVerifiedOutreachBatch(options.contacts);
   112	  const snapshot = options.snapshot ?? {
   113	    activeProspects: await db.select({
   114	      companyName: prospectsTable.companyName,
   115	      contactEmail: prospectsTable.contactEmail,
   116	      website: prospectsTable.website,
   117	      dedupeKey: prospectsTable.dedupeKey,
   118	      contactStatus: prospectsTable.contactStatus,
   119	    }).from(prospectsTable),
   120	    suppressedEmails: (await db.select({ email: outreachSuppressionsTable.email })
   121	      .from(outreachSuppressionsTable)).map((row) => row.email),
   122	  };
   123	  const report = evaluateVerifiedOutreachPreflight({
   124	    label: options.label,
   125	    target: options.target,
   126	    contacts: options.contacts,
   127	    activeProspects: snapshot.activeProspects,
   128	    suppressedEmails: snapshot.suppressedEmails,
   129	  });
   130	  if (options.enforceTarget && report.replacementsNeeded > 0) {
   131	    const blocked = report.conflicts
   132	      .map((conflict) => `${conflict.companyName} (${conflict.conflicts.join(", ")})`)
   133	      .join("; ");
   134	    throw new Error(
   135	      `${report.label} preflight requires ${report.target} usable contacts; `
   136	      + `found ${report.usableCount}, replacements needed ${report.replacementsNeeded}. `
   137	      + `Blocked companies: ${blocked}`,
   138	    );
   139	  }
   140	  return report;
   141	}

   143	export async function seedVerifiedOutreachBatch(options: {
   144	  enabled?: boolean;
   145	  now?: Date;
   146	} = {}): Promise<{ state: "skipped" | "ready"; queued: number }> {
   147	  const enabled = options.enabled ?? process.env.OUTREACH_SEED_VERIFIED_BATCH === "true";
   148	  if (!enabled) return { state: "skipped", queued: 0 };
   149	  assertVerifiedOutreachBatch(VERIFIED_OUTREACH_CONTACTS, 313);
   150	  assertVerifiedOutreachBatch(VERIFIED_OUTREACH_CONTACTS_AUG_30);
   151	  assertVerifiedOutreachBatch(VERIFIED_OUTREACH_CONTACTS_AUG_31, AUG_31_TARGET);
   152	  assertVerifiedOutreachBatch(VERIFIED_OUTREACH_CONTACTS_SEP_02, SEP_02_TARGET);
   153	  assertVerifiedOutreachBatch(
   154	    VERIFIED_OUTREACH_CONTACTS_SEP_02_PUBLIC,
   155	    SEP_02_PUBLIC_TARGET,
   156	  );
   157	  assertVerifiedOutreachBatch(
   158	    VERIFIED_OUTREACH_CONTACTS_SEP_05_DIRECT,
   159	    SEP_05_DIRECT_LIBRARY_TARGET,
   160	  );
   161	  assertVerifiedOutreachBatch(
   162	    VERIFIED_OUTREACH_CONTACTS_SEP_05,
   163	    SEP_05_PUBLIC_TARGET,
   164	  );
   165	
   166	  await preflightVerifiedOutreachBatch({
   167	    contacts: VERIFIED_OUTREACH_CONTACTS_SEP_05_DIRECT,
   168	    target: SEP_05_DIRECT_REQUIRED_TARGET,
   169	    label: "September 5",
   170	    enforceTarget: true,
   171	  });
   172	  await preflightVerifiedOutreachBatch({
   173	    contacts: VERIFIED_OUTREACH_CONTACTS_SEP_05,
   174	    target: SEP_05_PUBLIC_TARGET,
   175	    label: "September 5 Public",
   176	    enforceTarget: true,
   177	  });
   178	
   179	  let [campaign] = await db.select().from(campaignsTable)
   180	    .where(eq(campaignsTable.name, CAMPAIGN_NAME))
   181	    .limit(1);
   182	  if (!campaign) {
   183	    [campaign] = await db.insert(campaignsTable).values({
   184	      name: CAMPAIGN_NAME,
   185	      audience: "mixed",
   186	      states: [...LICENSED_OUTREACH_STATES],
   187	      dailyLimit: REGULAR_OUTREACH_DAILY_TARGET,
   188	      status: "active",
   189	      subjectTemplate: approvedOutreachSubject(),
   190	      bodyTemplate: "Approved personalized Apex Grid outreach copy",
   191	    }).returning();
   192	  }
   193	  if (!campaign) throw new Error("Unable to create the verified outreach campaign");
   194	  if (
   195	    campaign.dailyLimit !== REGULAR_OUTREACH_DAILY_TARGET
   196	    || campaign.audience !== "mixed"
   197	    || campaign.states.length !== LICENSED_OUTREACH_STATES.length
   198	    || LICENSED_OUTREACH_STATES.some((state) => !campaign.states.includes(state))
   199	  ) {
   200	    const [updatedCampaign] = await db.update(campaignsTable)
   201	      .set({
   202	        dailyLimit: REGULAR_OUTREACH_DAILY_TARGET,
   203	        audience: "mixed",
   204	        states: [...LICENSED_OUTREACH_STATES],
   205	      })
   206	      .where(eq(campaignsTable.id, campaign.id))
   207	      .returning();
   208	    if (!updatedCampaign) throw new Error("Unable to set the outreach campaign daily limit");
   209	    campaign = updatedCampaign;
   210	  }
   211	  await db.update(outreachResearchSchedulesTable)
   212	    .set({ targetCount: 150 })
   213	    .where(eq(outreachResearchSchedulesTable.campaignId, campaign.id));
   214	
   215	  const existingLegacyRows = await db.select({ dedupeKey: prospectsTable.dedupeKey })
   216	    .from(prospectsTable)
   217	    .where(inArray(
   218	      prospectsTable.dedupeKey,
   219	      VERIFIED_OUTREACH_CONTACTS.map((contact) => contact.dedupeKey),
   220	    ));
   221	  const existingLegacyKeys = new Set(
   222	    existingLegacyRows.flatMap((row) => row.dedupeKey ? [row.dedupeKey] : []),
   223	  );
   224	
   225	  for (const contact of VERIFIED_OUTREACH_CONTACTS) {
   226	    if (existingLegacyKeys.has(contact.dedupeKey)) continue;
   227	    await db.transaction(async (tx) => {
   228	      const normalizedEmail = contact.contactEmail.toLowerCase();
   229	      await tx.execute(sql`select pg_advisory_xact_lock(hashtextextended(${normalizedEmail}, 0))`);
   230	      const [existingSuppression] = await tx.select({ id: outreachSuppressionsTable.id })
   231	        .from(outreachSuppressionsTable)
   232	        .where(eq(outreachSuppressionsTable.email, normalizedEmail))
   233	        .limit(1);
   234	      if (existingSuppression) return;
   235	
   236	      const [prospect] = await tx.insert(prospectsTable).values({
   237	      campaignId: campaign.id,
   238	      companyName: contact.companyName,
   239	      website: contact.website,
   240	      city: contact.city,
   241	      state: contactState(contact),
   242	      audience: contact.audience,
   243	      sourceUrl: contact.sourceUrl,
   244	      researchNotes: contact.needSignals,
   245	      fitScore: 85,
   246	      needScore: 80,
   247	      needSignals: contact.needSignals,
   248	      contactName: contact.contactName,
   249	      contactTitle: contact.contactTitle,
   250	      contactEmail: normalizedEmail,
   251	      contactConfidence: "high",
   252	      contactSourceUrl: contact.contactSourceUrl,
   253	      dedupeKey: contact.dedupeKey,
   254	      emailStatus: "verified",
   255	      status: "approved",
   256	    }).onConflictDoUpdate({
   257	      target: prospectsTable.dedupeKey,
   258	      set: {
   259	        campaignId: campaign.id,
   260	         state: contactState(contact),
   261	        researchNotes: contact.needSignals,
   262	        needSignals: contact.needSignals,
   263	        contactName: contact.contactName,
   264	        contactTitle: contact.contactTitle,
   265	        contactEmail: normalizedEmail,
   266	        contactConfidence: "high",
   267	        contactSourceUrl: contact.contactSourceUrl,
   268	        emailStatus: "verified",
   269	        status: sql`case when ${prospectsTable.status} = 'contacted' then ${prospectsTable.status} else 'approved' end`,
   270	        updatedAt: new Date(),
   271	      },
   272	      setWhere: and(
   273	        eq(prospectsTable.contactStatus, "active"),
   274	        notInArray(prospectsTable.status, ["review", "replied", "not_a_fit", "suppressed"]),
   275	      ),
   276	    }).returning();
   277	      if (!prospect) return;
   278	
   279	    });
   280	  }
   281	  const existingRows = await db.select().from(prospectsTable);
   282	  const suppressions = await db.select({ email: outreachSuppressionsTable.email })
   283	    .from(outreachSuppressionsTable);
   284	  const suppressedEmails = new Set(suppressions.map((row) => row.email.trim().toLowerCase()));
   285	  const usedEmails = new Set(existingRows.flatMap((row) =>
   286	    row.contactEmail ? [row.contactEmail.trim().toLowerCase()] : []
   287	  ));
   288	  const usedDomains = new Set(existingRows.flatMap((row) => {
   289	    if (!row.website) return [];
   290	    try {
   291	      return [new URL(row.website).hostname.replace(/^www\./, "").toLowerCase()];
   292	    } catch {
   293	      return [];
   294	    }
   295	  }));
   296	  const datedBatches = [
   297	    {
   298	      contacts: VERIFIED_OUTREACH_CONTACTS_AUG_30,
   299	      prefix: "verified-2026-08-30-",
   300	      target: AUG_30_TARGET,
   301	      label: "August 30",
   302	      enforceTarget: false,
   303	    },
   304	    {
   305	      contacts: VERIFIED_OUTREACH_CONTACTS_AUG_31,
   306	      prefix: "verified-2026-08-31-",
   307	      target: AUG_31_TARGET,
   308	      label: "August 31",
   309	      enforceTarget: false,
   310	    },
   311	    {
   312	      contacts: VERIFIED_OUTREACH_CONTACTS_SEP_02,
   313	      prefix: "verified-2026-09-02-",
   314	      target: SEP_02_TARGET,
   315	      label: "September 2",
   316	      enforceTarget: false,
   317	    },
   318	    {
   319	      contacts: VERIFIED_OUTREACH_CONTACTS_SEP_02_PUBLIC,
   320	      prefix: "public-verified-2026-09-02-",
   321	      target: SEP_02_PUBLIC_TARGET,
   322	      label: "September 2 Public",
   323	      enforceTarget: false,
   324	    },
   325	    {
   326	      contacts: VERIFIED_OUTREACH_CONTACTS_SEP_05_DIRECT,
   327	      prefix: "verified-2026-09-05-",
   328	      target: SEP_05_DIRECT_REQUIRED_TARGET,
   329	      label: "September 5",
   330	      enforceTarget: true,
   331	    },
   332	    {
   333	      contacts: VERIFIED_OUTREACH_CONTACTS_SEP_05,
   334	      prefix: "public-verified-2026-09-05-",
   335	      target: SEP_05_PUBLIC_TARGET,
   336	      label: "September 5 Public",
   337	      enforceTarget: true,
   338	    },
   339	  ] as const;
   340	
   341	  for (const batch of datedBatches) {
   342	    let stored = existingRows.filter((row) =>
   343	      row.dedupeKey?.startsWith(batch.prefix)
   344	      && row.status === "approved"
   345	      && row.emailStatus === "verified"
   346	      && row.contactStatus === "active"
   347	    ).length;
   348	
   349	    for (const contact of batch.contacts) {
   350	      if (stored >= batch.target) break;
   351	    const normalizedEmail = contact.contactEmail.toLowerCase();
   352	    const domain = new URL(contact.website).hostname.replace(/^www\./, "").toLowerCase();
   353	    if (
   354	      suppressedEmails.has(normalizedEmail)
   355	      || usedEmails.has(normalizedEmail)
   356	      || usedDomains.has(domain)
   357	    ) continue;
   358	
   359	    const inserted = await db.transaction(async (tx) => {
   360	      await tx.execute(sql`select pg_advisory_xact_lock(hashtextextended(${normalizedEmail}, 0))`);
   361	      await tx.execute(sql`select pg_advisory_xact_lock(hashtextextended(${domain}, 0))`);
   362	      const currentRows = await tx.select({
   363	        contactEmail: prospectsTable.contactEmail,
   364	        website: prospectsTable.website,
   365	      }).from(prospectsTable);
   366	      const conflict = currentRows.some((row) => {
   367	        const email = row.contactEmail?.trim().toLowerCase();
   368	        let currentDomain = "";
   369	        try {
   370	          currentDomain = row.website
   371	            ? new URL(row.website).hostname.replace(/^www\./, "").toLowerCase()
   372	            : "";
   373	        } catch {
   374	          currentDomain = "";
   375	        }
   376	        return email === normalizedEmail || currentDomain === domain;
   377	      });
   378	      const [suppression] = await tx.select({ id: outreachSuppressionsTable.id })
   379	        .from(outreachSuppressionsTable)
   380	        .where(eq(outreachSuppressionsTable.email, normalizedEmail))
   381	        .limit(1);
   382	      if (conflict || suppression) return false;
   383	
   384	      const [prospect] = await tx.insert(prospectsTable).values({
   385	        campaignId: campaign.id,
   386	        companyName: contact.companyName,
   387	        website: contact.website,
   388	        city: contact.city,
   389	        state: contact.state,
   390	        audience: contact.audience,
   391	        sourceUrl: contact.sourceUrl,
   392	        researchNotes: `${contact.needSignals} Approval: ${contact.approvalStatus}.`,
   393	        fitScore: 85,
   394	        needScore: 80,
   395	        needSignals: contact.needSignals,
   396	        contactName: contact.contactName,
   397	        contactTitle: contact.contactTitle,
   398	        contactEmail: normalizedEmail,
   399	        contactConfidence: "high",
   400	        contactSourceUrl: contact.contactSourceUrl,
   401	        dedupeKey: contact.dedupeKey,
   402	        emailStatus: "verified",
   403	        status: "approved",
   404	        contactStatus: "active",
   405	        contactEvidenceType: contact.emailLane === "personal"
   406	          ? "findymail_verified"
   407	          : "official_publication",
   408	        contactEvidence: contact.emailEvidence,
   409	        contactEvidenceAt: options.now ?? new Date(),
   410	      }).onConflictDoNothing().returning({ id: prospectsTable.id });
   411	      return Boolean(prospect);
   412	    });
   413	    if (!inserted) continue;
   414	    usedEmails.add(normalizedEmail);
   415	    usedDomains.add(domain);
   416	    stored += 1;
   417	    }
   418	    if (batch.enforceTarget && stored < batch.target) {
   419	      throw new Error(
   420	        `Verified ${batch.label} outreach seed requires at least ${batch.target} active prospects; found ${stored}`,
   421	      );
   422	    }
   423	  }
   424	  // The seed only supplies reviewed candidates. The preparation service owns all
   425	  // message creation so the startup path cannot bypass the shared 200-slot ledger.
   426	  return { state: "ready", queued: 0 };
   427	}

   429	function contactState(contact: unknown): string {
   430	  if (
   431	    typeof contact === "object"
   432	    && contact !== null
   433	    && "state" in contact
   434	    && typeof contact.state === "string"
   435	  ) return contact.state;
   436	  return "AZ";
