     1	import { and, asc, eq, inArray, sql } from "drizzle-orm";
     2	import {
     3	  db,
     4	  outreachCatchUpCohortsTable,
     5	  outreachCatchUpEnrollmentsTable,
     6	  outreachCatchUpReservationsTable,
     7	  outreachProviderHandoffsTable,
     8	  outreachDeliveryEventsTable,
     9	  outreachMessagesTable,
    10	} from "@workspace/db";
    11	
    12	export const APEX_GRID_CATCH_UP_TARGET = 2_207;
    13	const ACCEPTANCE_EVENTS = ["processed", "deferred", "delivered"] as const;
    14	
    15	export type CatchUpProgress = {
    16	  target: number;
    17	  enrolled: number;
    18	  remaining: number;
    19	  status: string;
    20	};
    21	
    22	/** Counts only idempotently enrolled sequence-one messages with provider evidence. */
    23	export async function getCatchUpProgress(): Promise<CatchUpProgress> {
    24	  const [cohort] = await db.select().from(outreachCatchUpCohortsTable)
    25	    .where(eq(outreachCatchUpCohortsTable.name, "Apex Grid fresh verified catch-up")).limit(1);
    26	  if (!cohort) return { target: APEX_GRID_CATCH_UP_TARGET, enrolled: 0, remaining: APEX_GRID_CATCH_UP_TARGET, status: "missing" };
    27	  const [row] = await db.select({ count: sql<number>`count(*)::int` })
    28	    .from(outreachCatchUpEnrollmentsTable)
    29	    .where(eq(outreachCatchUpEnrollmentsTable.cohortId, cohort.id));
    30	  const enrolled = Math.min(row?.count ?? 0, cohort.targetCount);
    31	  return { target: cohort.targetCount, enrolled, remaining: Math.max(0, cohort.targetCount - enrolled), status: cohort.status };
    32	}

    34	/**
    35	 * Attributes one provider-accepted opener. The cohort row is locked before
    36	 * counting/inserting, so concurrent webhook/reconciliation workers cannot
    37	 * cross the target. Duplicate provider events and retries are harmless.
    38	 */
    39	export async function enrollAcceptedCatchUpMessage(messageId: number): Promise<boolean> {
    40	  return db.transaction(async (tx) => {
    41	    const [cohort] = await tx.select().from(outreachCatchUpCohortsTable)
    42	      .where(and(
    43	        eq(outreachCatchUpCohortsTable.name, "Apex Grid fresh verified catch-up"),
    44	        eq(outreachCatchUpCohortsTable.status, "active"),
    45	      )).for("update");
    46	    if (!cohort) return false;
    47	    const [message] = await tx.select().from(outreachMessagesTable)
    48	      .where(and(
    49	        eq(outreachMessagesTable.id, messageId),
    50	        eq(outreachMessagesTable.catchUpCohortId, cohort.id),
    51	        eq(outreachMessagesTable.sequenceNumber, 1),
    52	      )).limit(1);
    53	    if (!message || message.createdAt < cohort.activatedAt) return false;
    54	    const [existing] = await tx.select({ id: outreachCatchUpEnrollmentsTable.id })
    55	      .from(outreachCatchUpEnrollmentsTable)
    56	      .where(eq(outreachCatchUpEnrollmentsTable.messageId, messageId)).limit(1);
    57	    if (existing) return false;
    58	    const [countRow] = await tx.select({ count: sql<number>`count(*)::int` })
    59	      .from(outreachCatchUpEnrollmentsTable).where(eq(outreachCatchUpEnrollmentsTable.cohortId, cohort.id));
    60	    if ((countRow?.count ?? 0) >= cohort.targetCount) {
    61	      await tx.update(outreachCatchUpCohortsTable).set({ status: "completed", completedAt: new Date() })
    62	        .where(eq(outreachCatchUpCohortsTable.id, cohort.id));
    63	      return false;
    64	    }
    65	    const evidenceRows = await tx.select().from(outreachDeliveryEventsTable)
    66	      .where(and(
    67	        eq(outreachDeliveryEventsTable.outreachMessageId, messageId),
    68	        inArray(outreachDeliveryEventsTable.eventType, [...ACCEPTANCE_EVENTS]),
    69	      )).orderBy(asc(outreachDeliveryEventsTable.occurredAt));
    70	    const evidence = evidenceRows.find((event) =>
    71	      event.eventType === "processed"
    72	      || event.eventType === "delivered"
    73	      || (
    74	        event.eventType === "deferred"
    75	        && evidenceRows.some((prior) => prior.eventType === "processed"
    76	          && prior.occurredAt <= event.occurredAt)
    77	      ));
    78	    const [handoff] = await tx.select().from(outreachProviderHandoffsTable)
    79	      .where(eq(outreachProviderHandoffsTable.messageId, messageId)).limit(1);
    80	    const acceptedAt = evidence?.occurredAt ?? handoff?.acceptedAt;
    81	    if (!acceptedAt) return false;
    82	    const [inserted] = await tx.insert(outreachCatchUpEnrollmentsTable).values({
    83	      cohortId: cohort.id,
    84	      messageId,
    85	      sequenceNumber: 1,
    86	      providerAcceptedAt: acceptedAt,
    87	      providerEvidenceEventId: evidence?.id ?? null,
    88	    }).onConflictDoNothing().returning({ id: outreachCatchUpEnrollmentsTable.id });
    89	    if (!inserted) return false;
    90	    await tx.update(outreachCatchUpReservationsTable).set({ status: "accepted" })
    91	      .where(eq(outreachCatchUpReservationsTable.messageId, messageId));
    92	    if ((countRow?.count ?? 0) + 1 >= cohort.targetCount) {
    93	      await tx.update(outreachCatchUpCohortsTable).set({ status: "completed", completedAt: new Date() })
    94	        .where(eq(outreachCatchUpCohortsTable.id, cohort.id));
    95	    }
    96	    return true;
    97	  });
    98	}

   100	export async function getCatchUpRemainingCapacity(): Promise<number> {
   101	  const progress = await getCatchUpProgress();
   102	  if (progress.status !== "active") return 0;
   103	  const [capacityUsed] = await db.select({ count: sql<number>`count(*)::int` })
   104	    .from(outreachCatchUpReservationsTable)
   105	    .where(and(eq(outreachCatchUpReservationsTable.cohortId, (
   106	      await db.select({ id: outreachCatchUpCohortsTable.id }).from(outreachCatchUpCohortsTable)
   107	        .where(eq(outreachCatchUpCohortsTable.name, "Apex Grid fresh verified catch-up")).limit(1)
   108	    )[0]?.id ?? -1), inArray(outreachCatchUpReservationsTable.status, ["reserved", "accepted"])));
   109	  return Math.max(0, progress.target - (capacityUsed?.count ?? 0));
   110	}
