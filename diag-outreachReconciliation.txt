    26	type SendGridActivityMessage = {
    27	  msg_id?: string;
    28	  status?: string;
    29	  last_event_time?: string;
    30	  events?: Array<string | { event_name?: string; event?: string; reason?: string; response?: string }>;

    32	type SendGridActivityEvent = NonNullable<SendGridActivityMessage["events"]>[number];

    38	export type ProviderActivityOutcome =
    39	  | {
    40	      state: "accepted";
    41	      providerMessageId: string | null;
    42	      occurredAt: Date;
    43	      delivered: boolean;
    44	      reason: string;
    45	    }
    46	  | {
    47	      state: "not_accepted";
    48	      providerMessageId: string | null;
    49	      occurredAt: Date;
    50	      reason: string;
    51	    }
    52	  | {
    53	      state: "terminal_failure";
    54	      providerMessageId: string | null;
    55	      occurredAt: Date;
    56	      eventType: "bounce" | "blocked" | "dropped";
    57	      reason: string;
    58	    }
    59	  | {
    60	      state: "ambiguous";
    61	      reason: string;
    62	    };

    74	  waiting: number;

    77	function eventName(event: SendGridActivityEvent): string {
    78	  return (typeof event === "string" ? event : event.event_name ?? event.event ?? "")
    79	    .trim()
    80	    .toLowerCase();
    81	}
    82	
    83	function eventReason(event: SendGridActivityEvent): string | null {
    84	  if (typeof event === "string") return null;
    85	  return (event.reason ?? event.response ?? "").trim() || null;
    86	}

    88	function parseOccurredAt(value: string | undefined): Date {
    89	  const parsed = value ? new Date(value) : new Date();
    90	  return Number.isNaN(parsed.getTime()) ? new Date() : parsed;
    91	}

    93	export function classifySendGridActivity(
    94	  activity: SendGridActivityResponse,
    95	): ProviderActivityOutcome {
    96	  const messages = activity.messages ?? [];
    97	  if (messages.length === 0) {
    98	    return {
    99	      state: "ambiguous",
   100	      reason: "SendGrid activity has no matching record yet",
   101	    };
   102	  }
   103	
   104	  const latest = [...messages].sort((left, right) =>
   105	    parseOccurredAt(right.last_event_time).getTime() - parseOccurredAt(left.last_event_time).getTime()
   106	  )[0]!;
   107	  const status = latest.status?.trim().toLowerCase() ?? "";
   108	  const events = latest.events ?? [];
   109	  const names = events.map(eventName);
   110	  const occurredAt = parseOccurredAt(latest.last_event_time);
   111	  const providerMessageId = latest.msg_id?.trim() || null;
   112	
   113	  if (status === "not_delivered" || names.some((name) => REJECTED_EVENT_TYPES.has(name))) {
   114	    const rejection = events.find((event) => REJECTED_EVENT_TYPES.has(eventName(event)));
   115	    const rejectionType = rejection ? eventName(rejection) : "dropped";
   116	    return {
   117	      state: "terminal_failure",
   118	      providerMessageId,
   119	      occurredAt,
   120	      eventType: REJECTED_EVENT_TYPES.has(rejectionType)
   121	        ? rejectionType as "bounce" | "blocked" | "dropped"
   122	        : "dropped",
   123	      reason: rejection
   124	        ? eventReason(rejection) ?? "SendGrid activity confirms non-delivery"
   125	        : "SendGrid activity confirms non-delivery",
   126	    };
   127	  }
   128	  if (status === "delivered" || names.includes("delivered")) {
   129	    return {
   130	      state: "accepted",
   131	      providerMessageId,
   132	      occurredAt,
   133	      delivered: true,
   134	      reason: "SendGrid activity confirms delivery",
   135	    };
   136	  }
   137	  if (
   138	    ["processed", "processing"].includes(status)
   139	    || names.some((name) => ACCEPTED_EVENT_TYPES.has(name))
   140	  ) {
   141	    return {
   142	      state: "accepted",
   143	      providerMessageId,
   144	      occurredAt,
   145	      delivered: false,
   146	      reason: "SendGrid activity confirms provider acceptance",
   147	    };
   148	  }
   149	  return {
   150	    state: "ambiguous",
   151	    reason: `SendGrid activity returned unresolved status "${status || "unknown"}"`,
   152	  };
   153	}

   155	function escapeSendGridQueryValue(value: string): string {
   156	  return value.replaceAll("\\", "\\\\").replaceAll('"', '\\"');
   157	}

   159	async function requestSendGridActivity(path: string): Promise<Response> {
   160	  const dedicatedSendGridKey = process.env.SENDGRID_ISOLATION_VERIFIED === "true"
   161	    ? process.env.SENDGRID_DEDICATED_API_KEY?.trim()
   162	    : undefined;
   163	  const sendgridSubuser = process.env.SENDGRID_SUBUSER_USERNAME?.trim();
   164	  if (dedicatedSendGridKey) {
   165	    return fetch(`https://api.sendgrid.com${path}`, {
   166	      method: "GET",
   167	      headers: { Authorization: `Bearer ${dedicatedSendGridKey}` },
   168	      signal: AbortSignal.timeout(SENDGRID_ACTIVITY_TIMEOUT_MS),
   169	    });
   170	  }
   171	  return new ReplitConnectors().proxy("sendgrid", path, {
   172	    method: "GET",
   173	    headers: sendgridSubuser ? { "on-behalf-of": sendgridSubuser } : undefined,
   174	  });
   175	}

   177	export const lookupSendGridActivity: SendGridActivityLookup = async (reconciliationKey) => {
   178	  const query = `(unique_args['outreach_reconciliation_key']="${escapeSendGridQueryValue(reconciliationKey)}")`;
   179	  const path = `/v3/messages?limit=10&query=${encodeURIComponent(query)}`;
   180	  let response: Response;
   181	  try {
   182	    response = await Promise.race([
   183	      requestSendGridActivity(path),
   184	      new Promise<never>((_resolve, reject) => {
   185	        const timer = setTimeout(
   186	          () => reject(new Error("request timed out")),
   187	          SENDGRID_ACTIVITY_TIMEOUT_MS,
   188	        );
   189	        timer.unref();
   190	      }),
   191	    ]);
   192	  } catch (error) {
   193	    const detail = error instanceof Error ? error.message : "request failed";
   194	    return {
   195	      state: "ambiguous",
   196	      reason: `SendGrid activity lookup timed out or failed: ${detail}`,
   197	    };
   198	  }
   199	  if (!response.ok) {
   200	    return {
   201	      state: "ambiguous",
   202	      reason: `SendGrid activity lookup returned status ${response.status}`,
   203	    };
   204	  }
   205	  try {
   206	    return classifySendGridActivity(await response.json() as SendGridActivityResponse);
   207	  } catch {
   208	    return {
   209	      state: "ambiguous",
   210	      reason: "SendGrid activity lookup returned an invalid response",
   211	    };
   212	  }
   213	};

   215	function classifyRecordedEvents(
   216	  events: OutreachDeliveryEvent[],
   217	): ProviderActivityOutcome | null {
   218	  const rejected = events.find((event) => REJECTED_EVENT_TYPES.has(event.eventType));
   219	  if (rejected) {
   220	    return {
   221	      state: "terminal_failure",
   222	      providerMessageId: rejected.providerMessageId,
   223	      occurredAt: rejected.occurredAt,
   224	      eventType: rejected.eventType as "bounce" | "blocked" | "dropped",
   225	      reason: rejected.reason ?? "Recorded SendGrid event confirms non-delivery",
   226	    };
   227	  }
   228	  const delivered = events.find((event) => event.eventType === "delivered");
   229	  if (delivered) {
   230	    return {
   231	      state: "accepted",
   232	      providerMessageId: delivered.providerMessageId,
   233	      occurredAt: delivered.occurredAt,
   234	      delivered: true,
   235	      reason: "Recorded SendGrid event confirms delivery",
   236	    };
   237	  }
   238	  const accepted = events.find((event) => ACCEPTED_EVENT_TYPES.has(event.eventType));
   239	  if (accepted) {
   240	    return {
   241	      state: "accepted",
   242	      providerMessageId: accepted.providerMessageId,
   243	      occurredAt: accepted.occurredAt,
   244	      delivered: false,
   245	      reason: "Recorded SendGrid event confirms provider acceptance",
   246	    };
   247	  }
   248	  return null;
   249	}

   251	async function applyAcceptedOutcome(
   252	  message: OutreachMessage,
   253	  outcome: Extract<ProviderActivityOutcome, { state: "accepted" }>,
   254	): Promise<boolean> {
   255	  const [updated] = await db.update(outreachMessagesTable)
   256	    .set({
   257	      status: outcome.delivered ? "delivered" : "sent",
   258	      sentAt: outcome.occurredAt,
   259	      providerMessageId: outcome.providerMessageId ?? undefined,
   260	      error: null,
   261	    })
   262	    .where(and(
   263	      eq(outreachMessagesTable.id, message.id),
   264	      inArray(outreachMessagesTable.status, ["sending", "needs_review"]),
   265	    ))
   266	    .returning({ id: outreachMessagesTable.id });
   267	  // Enrollment still requires immutable HTTP-handoff or provider-event
   268	  // evidence. This lets a needs_review row recover when that evidence exists
   269	  // without admitting generic ambiguous historical attempts.
   270	  if (updated && message.catchUpCohortId) await enrollAcceptedCatchUpMessage(message.id);
   271	  return Boolean(updated);
   272	}

   274	async function applyRejectedOutcome(
   275	  message: OutreachMessage,
   276	  email: string,
   277	  outcome: Extract<ProviderActivityOutcome, { state: "not_accepted" }>,
   278	): Promise<"retry_released" | "failed" | "unchanged"> {
   279	  return db.transaction(async (tx) => {
   280	    const [current] = await tx.select({ status: outreachMessagesTable.status })
   281	      .from(outreachMessagesTable)
   282	      .where(eq(outreachMessagesTable.id, message.id))
   283	      .for("update");
   284	    if (!current || !["sending", "needs_review"].includes(current.status)) return "unchanged";
   285	
   286	    const [priorRetry] = await tx.select({ id: outreachDeliveryEventsTable.id })
   287	      .from(outreachDeliveryEventsTable)
   288	      .where(and(
   289	        eq(outreachDeliveryEventsTable.outreachMessageId, message.id),
   290	        eq(outreachDeliveryEventsTable.eventType, RETRY_RELEASE_EVENT_TYPE),
   291	      ))
   292	      .limit(1);
   293	
   294	    await Promise.all([
   295	      tx.delete(outreachSendReservationsTable)
   296	        .where(eq(outreachSendReservationsTable.messageId, message.id)),
   297	      tx.delete(outreachMonthlySendReservationsTable)
   298	        .where(eq(outreachMonthlySendReservationsTable.messageId, message.id)),
   299	      tx.delete(outreachSequenceSendClaimsTable)
   300	        .where(eq(outreachSequenceSendClaimsTable.messageId, message.id)),
   301	    ]);
   302	
   303	    if (priorRetry) {
   304	      await tx.update(outreachMessagesTable)
   305	        .set({
   306	          status: "failed",
   307	          providerMessageId: outcome.providerMessageId ?? undefined,
   308	          error: `SendGrid confirmed non-delivery after the one automatic retry: ${outcome.reason}`,
   309	        })
   310	        .where(eq(outreachMessagesTable.id, message.id));
   311	      await tx.update(outreachCatchUpReservationsTable).set({ status: "released", releasedAt: outcome.occurredAt })
   312	        .where(and(
   313	          eq(outreachCatchUpReservationsTable.messageId, message.id),
   314	          eq(outreachCatchUpReservationsTable.status, "reserved"),
   315	        ));
   316	      return "failed";
   317	    }
   318	
   319	    await tx.insert(outreachDeliveryEventsTable).values({
   320	      providerMessageId: outcome.providerMessageId,
   321	      email,
   322	      eventType: RETRY_RELEASE_EVENT_TYPE,
   323	      reason: outcome.reason,
   324	      occurredAt: outcome.occurredAt,
   325	      outreachMessageId: message.id,
   326	      reconciliationKey: message.providerReconciliationKey,
   327	    });
   328	    await tx.update(outreachMessagesTable)
   329	      .set({
   330	        status: "approved",
   331	        scheduledAt: new Date(),
   332	        providerMessageId: null,
   333	        error: `SendGrid confirmed non-delivery; released for one automatic retry: ${outcome.reason}`,
   334	      })
   335	      .where(eq(outreachMessagesTable.id, message.id));
   336	    if (message.catchUpCohortId) {
   337	      await tx.update(outreachCatchUpReservationsTable).set({ status: "released", releasedAt: outcome.occurredAt })
   338	        .where(eq(outreachCatchUpReservationsTable.messageId, message.id));
   339	      await tx.execute(sql`select ${outreachCatchUpCohortsTable.id} from ${outreachCatchUpCohortsTable}
   340	        where ${outreachCatchUpCohortsTable.id} = ${message.catchUpCohortId} for update`);
   341	      const [reserved] = await tx.select({ count: sql<number>`count(*)::int` })
   342	        .from(outreachCatchUpReservationsTable)
   343	        .where(and(eq(outreachCatchUpReservationsTable.cohortId, message.catchUpCohortId),
   344	          eq(outreachCatchUpReservationsTable.status, "reserved")));
   345	      const [cohort] = await tx.select({ target: outreachCatchUpCohortsTable.targetCount })
   346	        .from(outreachCatchUpCohortsTable).where(eq(outreachCatchUpCohortsTable.id, message.catchUpCohortId));
   347	      if ((reserved?.count ?? 0) < (cohort?.target ?? 0)) {
   348	        await tx.update(outreachCatchUpReservationsTable).set({ status: "reserved", releasedAt: null })
   349	          .where(eq(outreachCatchUpReservationsTable.messageId, message.id));
   350	      } else {
   351	        await tx.update(outreachMessagesTable).set({ status: "needs_review" })
   352	          .where(eq(outreachMessagesTable.id, message.id));
   353	      }
   354	    }
   355	    return "retry_released";
   356	  });
   357	}

   359	async function applyTerminalFailure(
   360	  message: OutreachMessage,
   361	  email: string,
   362	  outcome: Extract<ProviderActivityOutcome, { state: "terminal_failure" }>,
   363	): Promise<boolean> {
   364	  await suppressOutreachEmail(email, outcome.eventType, {
   365	    pendingStatus: "bounced",
   366	    pendingError: outcome.reason,
   367	    emailStatus: "invalid",
   368	  });
   369	  const [updated] = await db.update(outreachMessagesTable)
   370	    .set({
   371	      status: "bounced",
   372	      providerMessageId: outcome.providerMessageId ?? undefined,
   373	      error: outcome.reason,
   374	    })
   375	    .where(and(
   376	      eq(outreachMessagesTable.id, message.id),
   377	      inArray(outreachMessagesTable.status, ["sending", "needs_review", "bounced"]),
   378	    ))
   379	    .returning({ id: outreachMessagesTable.id });
   380	  await db.update(outreachCatchUpReservationsTable).set({ status: "released", releasedAt: outcome.occurredAt })
   381	    .where(and(
   382	      eq(outreachCatchUpReservationsTable.messageId, message.id),
   383	      eq(outreachCatchUpReservationsTable.status, "reserved"),
   384	    ));
   385	  return Boolean(updated);
   386	}

   388	export async function reconcileUncertainOutreachMessages(options: {
   389	  now?: Date;
   390	  reviewAfterMs?: number;
   391	  maxLookups?: number;
   392	  lookupActivity?: SendGridActivityLookup;
   393	  candidateMessageIds?: number[];
   394	} = {}): Promise<OutreachReconciliationSummary> {
   395	  const now = options.now ?? new Date();
   396	  const reviewAfterMs = options.reviewAfterMs ?? DEFAULT_REVIEW_AFTER_MS;
   397	  const maxLookups = options.maxLookups ?? DEFAULT_MAX_LOOKUPS;
   398	  const lookupActivity = options.lookupActivity ?? lookupSendGridActivity;
   399	  const summary: OutreachReconciliationSummary = {
   400	    checked: 0,
   401	    accepted: 0,
   402	    retryReleased: 0,
   403	    failed: 0,
   404	    ambiguous: 0,
   405	    waiting: 0,
   406	  };
   407	  const messages = await db.select({
   408	    message: outreachMessagesTable,
   409	    email: prospectsTable.contactEmail,
   410	  })
   411	    .from(outreachMessagesTable)
   412	    .innerJoin(prospectsTable, eq(outreachMessagesTable.prospectId, prospectsTable.id))
   413	    .where(options.candidateMessageIds === undefined
   414	      ? inArray(outreachMessagesTable.status, ["sending", "needs_review"])
   415	      : and(
   416	        inArray(outreachMessagesTable.status, ["sending", "needs_review"]),
   417	        inArray(outreachMessagesTable.id, options.candidateMessageIds),
   418	      ))
   419	    .orderBy(asc(outreachMessagesTable.updatedAt));
   420	
   421	  let lookupCount = 0;
   422	  for (const { message, email } of messages) {
   423	    const events = await db.select().from(outreachDeliveryEventsTable)
   424	      .where(or(
   425	        eq(outreachDeliveryEventsTable.outreachMessageId, message.id),
   426	        ...(message.providerReconciliationKey
   427	          ? [eq(outreachDeliveryEventsTable.reconciliationKey, message.providerReconciliationKey)]
   428	          : []),
   429	      ))
   430	      .orderBy(desc(outreachDeliveryEventsTable.occurredAt));
   431	    let outcome = classifyRecordedEvents(events);
   432	
   433	    if (!outcome) {
   434	      if (
   435	        message.status === "sending"
   436	        && now.getTime() - message.updatedAt.getTime() < reviewAfterMs
   437	      ) {
   438	        summary.waiting += 1;
   439	        continue;
   440	      }
   441	      summary.checked += 1;
   442	      if (!message.providerReconciliationKey) {
   443	        outcome = {
   444	          state: "ambiguous",
   445	          reason: "No stable SendGrid reconciliation key is stored; review provider activity manually",
   446	        };
   447	      } else if (lookupCount >= maxLookups) {
   448	        summary.waiting += 1;
   449	        summary.checked -= 1;
   450	        continue;
   451	      } else {
   452	        lookupCount += 1;
   453	        outcome = await lookupActivity(message.providerReconciliationKey);
   454	      }
   455	    } else {
   456	      summary.checked += 1;
   457	    }
   458	
   459	    if (outcome.state === "accepted") {
   460	      if (await applyAcceptedOutcome(message, outcome)) summary.accepted += 1;
   461	      continue;
   462	    }
   463	    if (outcome.state === "terminal_failure") {
   464	      if (await applyTerminalFailure(
   465	        message,
   466	        email?.trim().toLowerCase() || "unknown@example.invalid",
   467	        outcome,
   468	      )) {
   469	        summary.failed += 1;
   470	      }
   471	      continue;
   472	    }
   473	    if (outcome.state === "not_accepted") {
   474	      const applied = await applyRejectedOutcome(
   475	        message,
   476	        email?.trim().toLowerCase() || "unknown@example.invalid",
   477	        outcome,
   478	      );
   479	      if (applied === "retry_released") summary.retryReleased += 1;
   480	      if (applied === "failed") summary.failed += 1;
   481	      continue;
   482	    }
   483	
   484	    const [updated] = await db.update(outreachMessagesTable)
   485	      .set({
   486	        status: "needs_review",
   487	        error: `${outcome.reason}; retry remains blocked pending clear provider evidence`,
   488	      })
   489	      .where(and(
   490	        eq(outreachMessagesTable.id, message.id),
   491	        inArray(outreachMessagesTable.status, ["sending", "needs_review"]),
   492	      ))
   493	      .returning({ id: outreachMessagesTable.id });
   494	    if (updated) summary.ambiguous += 1;
   495	  }
   496	  return summary;
   497	}
