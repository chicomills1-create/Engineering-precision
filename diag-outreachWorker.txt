    81	function hasPublishedProductionUrl(): boolean {
    82	  const configuredUrl = process.env.PUBLIC_SITE_URL?.trim();
    83	  if (!configuredUrl || process.env.NODE_ENV !== "production") return false;
    84	
    85	  try {
    86	    const url = new URL(configuredUrl);
    87	    return url.protocol === "https:" && !url.username && !url.password;
    88	  } catch {
    89	    return false;
    90	  }
    91	}

    93	export function getOutreachAutomationStatus(): OutreachAutomationStatus {
    94	  const adminAllowlistReady = hasConfiguredAdminEmail();
    95	  const productionConfigReady = hasPublishedProductionUrl();
    96	  const dedicatedAccountReady = Boolean(process.env.SENDGRID_DEDICATED_API_KEY?.trim())
    97	    && process.env.SENDGRID_ISOLATION_VERIFIED === "true";
    98	  const sendgridDeliveryPathReady = Boolean(
    99	    (process.env.SENDGRID_SUBUSER_USERNAME?.trim()
   100	      && process.env.SENDGRID_SUBUSER_VERIFIED === "true")
   101	    || dedicatedAccountReady,
   102	  );
   103	  const deliveryEventsReady = Boolean(process.env.SENDGRID_EVENT_WEBHOOK_PUBLIC_KEY)
   104	    && process.env.SENDGRID_EVENT_PATH_VERIFIED === "true";
   105	  const replyWebhookReady = isReplyWebhookConfigured()
   106	    && Boolean(process.env.OUTREACH_REPLY_TO_EMAIL?.trim())
   107	    && process.env.OUTREACH_REPLY_PATH_VERIFIED === "true";
   108	  const automationEnabled = process.env.OUTREACH_AUTOMATION_ENABLED === "true";
   109	  const researchAutomationEnabled = process.env.OUTREACH_RESEARCH_AUTOMATION_ENABLED === "true";
   110	
   111	  return {
   112	    adminAllowlistReady,
   113	    productionConfigReady,
   114	    sendgridDeliveryPathReady,
   115	    deliveryEventsReady,
   116	    replyWebhookReady,
   117	    automationEnabled,
   118	    researchAutomationEnabled,
   119	    researchAutomationReady: adminAllowlistReady
   120	      && productionConfigReady
   121	      && researchAutomationEnabled,
   122	    automationReady: adminAllowlistReady
   123	      && productionConfigReady
   124	      && sendgridDeliveryPathReady
   125	      && deliveryEventsReady
   126	      && replyWebhookReady
   127	      && automationEnabled,
   128	  };
   129	}

   139	export async function claimOutreachMessageForSending(messageId: number): Promise<OutreachMessage | undefined> {
   140	  const [claimed] = await db.update(outreachMessagesTable)
   141	    .set({ status: "sending", error: null })
   142	    .where(and(
   143	      eq(outreachMessagesTable.id, messageId),
   144	      eq(outreachMessagesTable.status, "approved"),
   145	      sql`exists (
   146	        select 1
   147	        from ${prospectsTable} as current_prospect
   148	        where current_prospect.id = ${outreachMessagesTable.prospectId}
   149	          and current_prospect.contact_status = 'active'
   150	          and current_prospect.status in ('approved', 'contacted')
   151	          and not exists (
   152	            select 1
   153	            from outreach_suppressions as suppression
   154	            where suppression.email = lower(trim(current_prospect.contact_email))
   155	          )
   156	      )`,
   157	      or(
   158	        eq(outreachMessagesTable.sequenceNumber, 1),
   159	        sql`exists (
   160	          select 1
   161	          from ${outreachMessagesTable} as previous_message
   162	          where previous_message.prospect_id = ${outreachMessagesTable.prospectId}
   163	            and previous_message.sequence_number = ${outreachMessagesTable.sequenceNumber} - 1
   164	            and previous_message.status = 'delivered'
   165	            and (
   166	              previous_message.campaign_id = ${outreachMessagesTable.campaignId}
   167	              or (
   168	                previous_message.campaign_id is null
   169	                and ${outreachMessagesTable.campaignId} is null
   170	              )
   171	            )
   172	        )`,
   173	      ),
   174	      sql`${outreachMessagesTable.catchUpCohortId} is null or exists (
   175	        select 1 from outreach_catch_up_reservations as catch_up_reservation
   176	        where catch_up_reservation.message_id = ${outreachMessagesTable.id}
   177	          and catch_up_reservation.status = 'reserved'
   178	      )`,
   179	      or(
   180	        eq(outreachMessagesTable.sequenceNumber, 1),
   181	        sql`exists (
   182	          select 1
   183	          from ${outreachMessagesTable} as delivered_initial
   184	          inner join outreach_delivery_events as delivery_evidence
   185	            on delivery_evidence.outreach_message_id = delivered_initial.id
   186	          where delivered_initial.prospect_id = ${outreachMessagesTable.prospectId}
   187	            and delivered_initial.sequence_number = 1
   188	            and delivered_initial.status = 'delivered'
   189	            and delivery_evidence.event_type = 'delivered'
   190	            and (
   191	              delivered_initial.campaign_id = ${outreachMessagesTable.campaignId}
   192	              or (
   193	                delivered_initial.campaign_id is null
   194	                and ${outreachMessagesTable.campaignId} is null
   195	              )
   196	            )
   197	        )`,
   198	      ),
   199	      sql`${outreachMessagesTable.sequenceNumber} <= 2`,
   200	      or(
   201	        eq(outreachMessagesTable.sequenceNumber, 1),
   202	        sql`exists (
   203	          select 1
   204	          from ${outreachMessagesTable} as initial_message
   205	          inner join outreach_delivery_events as initial_engagement
   206	            on initial_engagement.outreach_message_id = initial_message.id
   207	          where initial_message.prospect_id = ${outreachMessagesTable.prospectId}
   208	            and initial_message.sequence_number = 1
   209	            and initial_engagement.event_type in ('open', 'click')
   210	             and initial_engagement.occurred_at >= now() - (${OPENER_FOLLOW_UP_MAX_AGE_DAYS} * interval '1 day')
   211	              and initial_engagement.occurred_at <= now()
   212	            and exists (
   213	              select 1
   214	              from outreach_delivery_events as delivery_evidence
   215	              where delivery_evidence.outreach_message_id = initial_message.id
   216	                and delivery_evidence.event_type = 'delivered'
   217	                and delivery_evidence.occurred_at <= initial_engagement.occurred_at
   218	            )
   219	            and (
   220	              initial_message.campaign_id = ${outreachMessagesTable.campaignId}
   221	              or (
   222	                initial_message.campaign_id is null
   223	                and ${outreachMessagesTable.campaignId} is null
   224	              )
   225	            )
   226	        )`,
   227	      ),
   228	    ))
   229	    .returning();
   230	  if (!claimed) {
   231	    // Only an opener still approved can have been rejected by eligibility
   232	    // checks; a concurrent claimant is already sending and keeps its slot.
   233	    await db.update(outreachCatchUpReservationsTable).set({
   234	      status: "released",
   235	      releasedAt: new Date(),
   236	    }).where(and(
   237	      eq(outreachCatchUpReservationsTable.messageId, messageId),
   238	      eq(outreachCatchUpReservationsTable.status, "reserved"),
   239	      sql`exists (
   240	        select 1 from outreach_messages rejected
   241	        where rejected.id = ${messageId} and rejected.status = 'approved'
   242	      )`,
   243	    ));
   244	  }
   245	  return claimed;
   246	}

   262	export async function sendClaimedOutreachMessage(
   263	  message: OutreachMessage,
   264	  options: Omit<OutreachSendOptions, "expectedPersistedStatus" | "afterProviderDispatch"> = {},
   265	): Promise<OutreachMessage | undefined> {
   266	  const [prospect] = await db.select().from(prospectsTable)
   267	    .where(eq(prospectsTable.id, message.prospectId));
   268	  if (!prospect) throw new Error("Prospect not found");
   269	  const [campaign] = message.campaignId
   270	    ? await db.select().from(campaignsTable).where(eq(campaignsTable.id, message.campaignId))
   271	    : [];
   272	  let persisted: OutreachMessage | undefined;
   273	  await sendApprovedOutreach(
   274	    { ...message, status: "approved" },
   275	    prospect,
   276	    campaign,
   277	    {
   278	      ...options,
   279	      deliverabilityReady: getOutreachAutomationStatus().sendgridDeliveryPathReady || Boolean(options.dispatch),
   280	      expectedPersistedStatus: "sending",
   281	      afterProviderDispatch: async (providerMessageId) => {
   282	        try {
   283	          if (message.catchUpCohortId) {
   284	            await db.insert(outreachProviderHandoffsTable).values({
   285	              messageId: message.id,
   286	              providerMessageId,
   287	              reconciliationKey: message.providerReconciliationKey ?? `message:${message.id}`,
   288	              acceptedAt: new Date(),
   289	            }).onConflictDoNothing();
   290	            await enrollAcceptedCatchUpMessage(message.id);
   291	          }
   292	          if (message.sequenceNumber === 1) {
   293	            await ensureApprovedFollowUpSequence(
   294	              message,
   295	              prospect,
   296	              new Date(),
   297	              { emailLockAlreadyHeld: true },
   298	            );
   299	          }
   300	          [persisted] = await db.update(outreachMessagesTable).set({
   301	            status: "sent",
   302	            sentAt: new Date(),
   303	            providerMessageId,
   304	            error: null,
   305	          }).where(and(
   306	            eq(outreachMessagesTable.id, message.id),
   307	            eq(outreachMessagesTable.status, "sending"),
   308	          )).returning();
   309	          if (!persisted) {
   310	            throw new Error("Accepted provider handoff could not be persisted as sent");
   311	          }
   312	        } catch (error) {
   313	          throw new Error(
   314	            "Provider dispatch result is unknown: SendGrid accepted the handoff but local persistence failed",
   315	            { cause: error },
   316	          );
   317	        }
   318	      },
   319	    },
   320	  );
   321	  return persisted;
   322	}

   338	export function createGuardedAsyncRun(task: () => Promise<void>): () => void {
   339	  let running = false;
   340	  return () => {
   341	    if (running) return;
   342	    running = true;
   343	    void task().finally(() => {
   344	      running = false;
   345	    });
   346	  };
   347	}

   378	export async function reconcileSendingOutreachMessages(
   379	  now = new Date(),
   380	  reviewAfterMs = 15 * 60 * 1000,
   381	  candidateMessageIds?: number[],
   382	): Promise<number> {
   383	  const result = await reconcileUncertainOutreachMessages({
   384	    now,
   385	    reviewAfterMs,
   386	    candidateMessageIds,
   387	  });
   388	  return result.accepted + result.retryReleased + result.failed + result.ambiguous;

   391	export async function processDueOutreachMessagesWithSummary(): Promise<OutreachDispatchSummary> {
   392	  const summary: OutreachDispatchSummary = {
   393	    claimed: 0,
   394	    providerAccepted: 0,
   395	    stopped: 0,
   396	    unresolved: 0,
   397	  };
   398	  if (!isOutreachAutomationReady()) return summary;
   399	  await stopLegacyAdditionalFollowUps();
   400	  await stopStaleOpenerFollowUps();
   401	  await backfillDeliveredFollowUpSequences();
   402	  const due = await db.select().from(outreachMessagesTable)
   403	    .where(and(
   404	      eq(outreachMessagesTable.status, "approved"),
   405	      lte(outreachMessagesTable.scheduledAt, new Date()),
   406	    ))
   407	    .orderBy(asc(outreachMessagesTable.scheduledAt))
   408	    .limit(MAX_SCHEDULED_MESSAGES_PER_RUN);
   409	  for (const message of due) {
   410	    const claimed = await claimOutreachMessageForSending(message.id);
   411	    if (!claimed) {
   412	      summary.stopped += 1;
   413	      continue;
   414	    }
   415	    summary.claimed += 1;
   416	    try {
   417	      const persisted = await sendClaimedOutreachMessage(claimed);
   418	      summary.providerAccepted = countPersistedOutreachSend(summary.providerAccepted, persisted);
   419	    } catch (err) {
   420	      const error = err instanceof Error ? err.message : "Scheduled send failed";
   421	      if (err instanceof DailySendLimitError || err instanceof MonthlySendLimitError) {
   422	        const scheduledAt = getNextPhoenixEightPm();
   423	        await db.update(outreachMessagesTable)
   424	          .set({
   425	            status: "approved",
   426	            scheduledAt,
   427	            error: `${err.message}; deferred to ${scheduledAt.toISOString()}`,
   428	          })
   429	          .where(and(
   430	            eq(outreachMessagesTable.id, claimed.id),
   431	            eq(outreachMessagesTable.status, "sending"),
   432	          ));
   433	        continue;
   434	      }
   435	      if (err instanceof ProviderRateLimitError) {
   436	        const retryAt = new Date(Date.now() + err.retryAfterMs);
   437	        await db.update(outreachMessagesTable)
   438	          .set({
   439	            status: "approved",
   440	            scheduledAt: retryAt,
   441	            error: `${err.message}; deferred to ${retryAt.toISOString()}`,
   442	          })
   443	          .where(and(
   444	            eq(outreachMessagesTable.id, claimed.id),
   445	            eq(outreachMessagesTable.status, "sending"),
   446	          ));
   447	        logger.warn({ messageId: claimed.id, retryAt, error: err.message }, "Provider rate limit deferred outreach without consuming quota");
   448	        continue;
   449	      }
   450	      await db.update(outreachMessagesTable)
   451	        .set({ status: getSendFailureStatus(err), error })
   452	        .where(and(
   453	          eq(outreachMessagesTable.id, claimed.id),
   454	          eq(outreachMessagesTable.status, "sending"),
   455	        ));
   456	      if (getSendFailureStatus(err) === "needs_review") summary.unresolved += 1;
   457	      if (getSendFailureStatus(err) === "failed") {
   458	        await releaseDefinitiveCatchUpReservation(claimed.id);
   459	      }
   460	      logger.warn({ messageId: claimed.id, error }, "Scheduled outreach send blocked or failed");
   461	    }
   462	  }
   463	  return summary;
   464	}

   481	function logReconciliationSummary(summary: OutreachReconciliationSummary): void {
   482	  if (summary.checked === 0 && summary.waiting === 0) return;
   483	  const context = {
   484	    checked: summary.checked,
   485	    accepted: summary.accepted,
   486	    retryReleased: summary.retryReleased,
   487	    failed: summary.failed,
   488	    ambiguous: summary.ambiguous,
   489	    waiting: summary.waiting,
   490	  };
   491	  if (summary.ambiguous > 0 || summary.failed > 0) {
   492	    logger.warn(context, "Outreach reconciliation left messages requiring review");
   493	  } else {
   494	    logger.info(context, "Outreach reconciliation completed");
   495	  }
   496	}

   498	export function startOutreachWorker(): void {
   499	  void syncSendGridInboundReplyWebhook()
   500	    .then(({ state, hostname }) => {
   501	      if (state !== "skipped") {
   502	        logger.info({ state, hostname }, "SendGrid inbound reply webhook synchronized");
   503	      }
   504	    })
   505	    .catch((err: unknown) => {
   506	      logger.error({ err }, "SendGrid inbound reply webhook synchronization failed");
   507	    });
   508	
   509	  const status = getOutreachAutomationStatus();
   510	  if (!status.automationReady && !status.researchAutomationReady) {
   511	    logger.info({
   512	      adminAllowlistReady: status.adminAllowlistReady,
   513	      productionConfigReady: status.productionConfigReady,
   514	      sendgridDeliveryPathReady: status.sendgridDeliveryPathReady,
   515	      deliveryEventsReady: status.deliveryEventsReady,
   516	      replyWebhookReady: status.replyWebhookReady,
   517	      automationEnabled: status.automationEnabled,
   518	      researchAutomationEnabled: status.researchAutomationEnabled,
   519	    }, "Outreach schedulers disabled until production safety checks pass");
   520	    return;
   521	  }
   522	
   523	  const runs = createOutreachWorkerRuns({
   524	    dispatch: async () => {
   525	      await processDueOutreachMessages()
   526	        .then((sentCount) => {
   527	          if (sentCount > 0) logger.info({ sentCount }, "Processed scheduled outreach messages");
   528	        })
   529	        .catch((err: unknown) => logger.error({ err }, "Outreach dispatch scheduler failed"));
   530	      const monitorResult = await monitorOverdueOutreachQueue();
   531	      if (monitorResult.state === "alerted") {
   532	        logger.error(monitorResult.summary, "Approved outreach remained queued past its scheduled grace period");
   533	      } else if (monitorResult.state === "failed") {
   534	        logger.error(
   535	          { ...monitorResult.summary, error: monitorResult.error },
   536	          "Overdue outreach admin alert failed",
   537	        );
   538	      }
   539	    },
   540	    reconcile: async () => {
   541	      await processOutreachReconciliation()
   542	        .then(async () => {
   543	          const preparation = await prepareNextPhoenixOutreach();
   544	          const hotLeads = await prepareNextPhoenixHotLeadOutreach();
   545	          if (preparation.state === "completed") {
   546	            logger.info({ ...preparation, hotLeads }, "Prepared next Phoenix outreach window");
   547	          } else if (preparation.state === "failed") {
   548	            logger.error({ ...preparation, hotLeads }, "Next Phoenix outreach preparation failed");
   549	          }
   550	        })
   551	        .catch((err: unknown) => logger.error({ err }, "Outreach reconciliation scheduler failed"));
   552	    },
   553	    research: async () => {
   554	      await ensureRecurringHotMarketResearchSchedule()
   555	        .then(() => Promise.all([
   556	          processDueOutreachResearchSchedules(),
   557	          processDueHotMarketResearch(),
   558	        ]))
   559	        .then(([completedCount, hotMarket]) => {
   560	          if (completedCount > 0) {
   561	            logger.info({ completedCount }, "Prepared scheduled outreach research lists");
   562	          }
   563	          if (hotMarket.state === "completed") {
   564	            logger.info(hotMarket, "Replenished the next verified hot-market window");
   565	          } else if (hotMarket.state === "failed") {
   566	            logger.error(hotMarket, "Verified hot-market replenishment failed");
   567	          }
   568	        })
   569	        .catch((err: unknown) => logger.error({ err }, "Outreach research scheduler failed"));
   570	    },
   571	    monitor: async () => {
   572	      const result = await monitorOverdueOutreachQueue();
   573	      if (result.state === "alerted") {
   574	        logger.error(result.summary, "Approved outreach remained queued past its scheduled grace period");
   575	      } else if (result.state === "failed") {

   577	      }
   578	    },
   579	  });
   580	  trafficTriggeredDispatch = runs.runDispatch;
