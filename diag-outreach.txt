    62	function isPublicBusinessInbox(prospect: Pick<Prospect, "contactEmail" | "contactName" | "contactEvidenceType">): boolean {
    63	  const local = prospect.contactEmail?.toLowerCase().split("@")[0] ?? "";
    64	  const normalizedName = prospect.contactName?.trim().toLowerCase().replace(/[^a-z]/g, "") ?? "";
    65	  return prospect.contactEvidenceType === "official_publication"
    66	    && (!normalizedName || normalizedName === local.replace(/[^a-z]/g, "") || /(?:office|team|desk|inquiries)$/i.test(prospect.contactName?.trim() ?? ""))
    67	    && PUBLIC_GREETING_PARTS.has(local);
    68	}

   118	/**
   119	 * Source metadata is authoritative for Hot Market.  Older regular records did
   120	 * not persist a source, so their lane is inferred from the prospect evidence
   121	 * at dispatch rather than treating them as personal by default.
   122	 */
   123	export function getOutreachDailyLane(
   124	  message: Pick<OutreachMessage, "sourceType">,
   125	  prospect: Pick<Prospect, "contactEmail" | "contactName" | "contactEvidenceType">,
   126	): OutreachDailyLane {
   127	  // Hot-lead attribution is fail-closed: only the persisted sourceType values
   128	  // below are accepted. Engagement rows in outreachHotLeads.ts have no
   129	  // sourceType/campaign field, so they must not be guessed into this lane.
   130	  if (message.sourceType === "hot_lead" || message.sourceType === "hot_lead_verified") return "hot_lead";
   131	  if (isHotMarketSourceType(message.sourceType)) return "hot_market";
   132	  return isEvidenceBackedPublicInbox(
   133	    prospect.contactEmail,
   134	    prospect.contactName,
   135	    prospect.contactEvidenceType,
   136	  ) ? "public" : "named";
   137	}

   175	export async function ensureProviderReconciliationKey(messageId: number): Promise<string> {
   176	  const fallbackKey = `outreach-message-${messageId}`;
   177	  const [row] = await db.update(outreachMessagesTable)
   178	    .set({
   179	      providerReconciliationKey: sql`coalesce(${outreachMessagesTable.providerReconciliationKey}, ${fallbackKey})`,
   180	    })
   181	    .where(eq(outreachMessagesTable.id, messageId))
   182	    .returning({ providerReconciliationKey: outreachMessagesTable.providerReconciliationKey });
   183	  if (!row?.providerReconciliationKey) {
   184	    throw new Error("Unable to establish an outreach provider reconciliation key");
   185	  }
   186	  return row.providerReconciliationKey;
   187	}

   189	function phoenixDateKey(date = new Date()): string {
   190	  const parts = new Intl.DateTimeFormat("en-US", {
   191	    timeZone: "America/Phoenix",
   192	    year: "numeric",
   193	    month: "2-digit",
   194	    day: "2-digit",
   195	  }).formatToParts(date);
   196	  const value = Object.fromEntries(parts.map((part) => [part.type, part.value]));
   197	  return `${value.year}-${value.month}-${value.day}`;
   198	}

   200	export function getPhoenixOutreachMonthKey(date = new Date()): string {
   201	  return phoenixDateKey(date).slice(0, 7);

   204	export function getOutreachMonthlyLimit(date = new Date()): number {
   205	  const monthKey = getPhoenixOutreachMonthKey(date);
   206	  if (monthKey <= "2026-09") return OUTREACH_MONTHLY_LIMIT;
   207	  if (monthKey === "2026-10") return 20_000;
   208	  if (monthKey === "2026-11") return 35_000;
   209	  if (monthKey === "2026-12") return 50_000;
   210	  return 50_000;
   211	}

   220	export function getGlobalOutreachDailyLimit(
   221	  date = new Date(),
   222	  hotMarketMessageCount = 0,
   223	): number {
   224	  const month = getPhoenixOutreachMonthKey(date);
   225	  if (month === "2026-09") {
   226	    return phoenixDateKey(date) < "2026-09-13"
   227	      ? SEPTEMBER_OUTREACH_TOTAL_TARGET
   228	      : SEPTEMBER_OUTREACH_RELAUNCH_DAILY_TARGET;
   229	  }
   230	  if (month > "2026-09") return SEPTEMBER_OUTREACH_RELAUNCH_DAILY_TARGET;
   231	  return Math.max(0, hotMarketMessageCount);
   232	}

   266	async function releaseOutreachReservations(reservations: OutreachReservationIds): Promise<void> {
   267	  await db.transaction(async (tx) => {
   268	    if (reservations.dailyReservationId !== undefined) {
   269	      await tx.delete(outreachSendReservationsTable)
   270	        .where(eq(outreachSendReservationsTable.id, reservations.dailyReservationId));
   271	    }
   272	    if (reservations.regularMonthlyReservationId !== undefined) {
   273	      await tx.delete(outreachMonthlySendReservationsTable)
   274	        .where(eq(outreachMonthlySendReservationsTable.id, reservations.regularMonthlyReservationId));
   275	    }
   276	    await tx.delete(outreachSequenceSendClaimsTable)
   277	      .where(eq(outreachSequenceSendClaimsTable.id, reservations.sequenceClaimId));
   278	  });
   279	}

   281	async function reserveOutreachSend(
   282	  message: OutreachMessage,
   283	  campaign: Campaign | undefined,
   284	  normalizedEmail: string,
   285	  deliverabilityReady?: boolean,
   286	): Promise<OutreachReservationIds> {
   287	  const now = new Date();
   288	  const dateKey = phoenixDateKey(now);
   289	  const monthKey = getPhoenixOutreachMonthKey(now);
   290	  const monthlyQuotaKey = `outreach-authoritative:${monthKey}`;
   291	  const dayStart = getPhoenixCalendarDayStart(now);
   292	  const monthStart = new Date(`${monthKey}-01T07:00:00.000Z`);
   293	  return db.transaction(async (tx) => {
   294	    const [lockedProspect] = await tx.select({
   295	      contactEmail: prospectsTable.contactEmail,
   296	      contactName: prospectsTable.contactName,
   297	      contactEvidenceType: prospectsTable.contactEvidenceType,
   298	    })
   299	      .from(prospectsTable)
   300	      .where(eq(prospectsTable.id, message.prospectId))
   301	      .for("update");
   302	    const lane = getOutreachDailyLane(message, lockedProspect ?? {
   303	      contactEmail: null, contactName: null, contactEvidenceType: null,
   304	    });
   305	    const dailyQuotaKey = `outreach-${lane}:${dateKey}`;
   306	    await tx.execute(sql`select pg_advisory_xact_lock(hashtextextended(${normalizedEmail}, 0))`);
   307	    // One global Phoenix-day lock serializes every lane and campaign. Lane
   308	    // locks remain useful for observability, but can never create extra quota.
   309	    await tx.execute(sql`select pg_advisory_xact_lock(hashtext(${`outreach-authoritative:${dateKey}`}))`);
   310	    await tx.execute(sql`select pg_advisory_xact_lock(hashtext(${dailyQuotaKey}))`);
   311	    await tx.execute(sql`select pg_advisory_xact_lock(hashtext(${monthlyQuotaKey}))`);
   312	
   313	    if ((lockedProspect?.contactEmail?.trim().toLowerCase() || null) !== normalizedEmail) {
   314	      throw new Error("Contact changed while dispatch was starting; please retry");
   315	    }
   316	    const runtime = await loadOutreachSystemConfig(now, tx);
   317	    const safeguards = runtime.policy.sendingSafeguards;
   318	    if (safeguards.requireDeliverabilityReady) {
   319	      const dedicatedReady = Boolean(process.env.SENDGRID_DEDICATED_API_KEY?.trim())
   320	        && process.env.SENDGRID_ISOLATION_VERIFIED === "true";
   321	      const subuserReady = Boolean(process.env.SENDGRID_SUBUSER_USERNAME?.trim())
   322	        && process.env.SENDGRID_SUBUSER_VERIFIED === "true";
   323	      if (deliverabilityReady !== true && !dedicatedReady && !subuserReady) {
   324	        throw new Error("Outreach deliverability readiness is unavailable; reservation refused");
   325	      }
   326	    }
   327	    if (safeguards.requireQualifiedInventory) {
   328	      if (!lockedProspect
   329	        || !["approved", "review"].includes((await tx.select({ status: prospectsTable.status }).from(prospectsTable)
   330	          .where(eq(prospectsTable.id, message.prospectId)).limit(1))[0]?.status ?? "")
   331	        || lockedProspect.contactEmail === null) {
   332	        throw new Error("No qualified unsuppressed prospect inventory remains");
   333	      }
   334	      const [suppressed] = await tx.select({ id: outreachSuppressionsTable.id })
   335	        .from(outreachSuppressionsTable).where(eq(outreachSuppressionsTable.email, normalizedEmail)).limit(1);
   336	      if (suppressed) throw new Error("No qualified unsuppressed prospect inventory remains");
   337	    }
   338	
   339	    const duplicateStatuses = DUPLICATE_EMAIL_SEQUENCE_STATUSES.filter(isDuplicateEmailSequenceStatus);
   340	    const [historicalDuplicate] = await tx.select({ id: outreachMessagesTable.id })
   341	      .from(outreachMessagesTable)
   342	      .innerJoin(prospectsTable, eq(outreachMessagesTable.prospectId, prospectsTable.id))
   343	      .where(and(
   344	        sql`lower(trim(${prospectsTable.contactEmail})) = ${normalizedEmail}`,
   345	        eq(outreachMessagesTable.sequenceNumber, message.sequenceNumber),
   346	        ne(outreachMessagesTable.id, message.id),
   347	        inArray(outreachMessagesTable.status, duplicateStatuses),
   348	      ))
   349	      .limit(1);
   350	    if (historicalDuplicate) {
   351	      throw new Error("This sequence is already active or was sent to this email address");
   352	    }
   353	    const [existingEmailReservation] = await tx.select({
   354	      id: outreachMonthlySendReservationsTable.id,
   355	    })
   356	      .from(outreachMonthlySendReservationsTable)
   357	      .where(and(
   358	        eq(outreachMonthlySendReservationsTable.normalizedEmail, normalizedEmail),
   359	        eq(outreachMonthlySendReservationsTable.sequenceNumber, message.sequenceNumber),
   360	      ))
   361	      .limit(1);
   362	    if (existingEmailReservation) {
   363	      throw new Error("This sequence is already reserved or was sent to this email address");
   364	    }
   365	
   366	    const [sequenceClaim] = await tx.insert(outreachSequenceSendClaimsTable)
   367	      .values({
   368	        messageId: message.id,
   369	        prospectId: message.prospectId,
   370	        campaignScope: message.campaignId ? `campaign:${message.campaignId}` : "standalone",
   371	        sequenceNumber: message.sequenceNumber,
   372	      })
   373	      .onConflictDoNothing()
   374	      .returning({ id: outreachSequenceSendClaimsTable.id });
   375	    if (!sequenceClaim) {
   376	      throw new Error("This campaign sequence is already reserved or was sent to this prospect");
   377	    }
   378	
   379	    let dailyReservationId: number | undefined;
   380	    if (message.sequenceNumber === 1) {
   381	      const effectiveAllocations = effectiveLaneAllocations(runtime.schedule, now);
   382	      const monthEnd = new Date(`${monthKey}-01T07:00:00.000Z`);
   383	      monthEnd.setUTCMonth(monthEnd.getUTCMonth() + 1);
   384	      const [sent] = await tx.select({ value: count() }).from(outreachMessagesTable)
   385	        .where(and(eq(outreachMessagesTable.sequenceNumber, 1), gte(outreachMessagesTable.sentAt, monthStart), lt(outreachMessagesTable.sentAt, monthEnd)));
   386	      const [reserved] = await tx.select({ value: count() }).from(outreachMonthlySendReservationsTable)
   387	        .innerJoin(outreachMessagesTable, eq(outreachMonthlySendReservationsTable.messageId, outreachMessagesTable.id))
   388	        .where(and(eq(outreachMonthlySendReservationsTable.quotaKey, monthlyQuotaKey), eq(outreachMessagesTable.sequenceNumber, 1), inArray(outreachMessagesTable.status, ["approved", "sending", "needs_review"])));
   389	      const monthlyTarget = Math.min(50_000, runtime.schedule.monthlyTarget);
   390	      const used = (sent?.value ?? 0) + (reserved?.value ?? 0);
   391	      const remainingMonth = Math.max(0, monthlyTarget - used);
   392	      const daysInMonth = new Date(Date.UTC(Number(monthKey.slice(0, 4)), Number(monthKey.slice(5)), 0)).getUTCDate();
   393	      let remainingDays = 0;
   394	      for (let day = Number(dateKey.slice(8)); day <= daysInMonth; day += 1) {
   395	        const weekday = new Date(Date.UTC(Number(monthKey.slice(0, 4)), Number(monthKey.slice(5, 7)) - 1, day)).getUTCDay();
   396	        if (weekday !== 0 && weekday !== 6) remainingDays += 1;
   397	      }
   398	       const dailyAllowance = monthKey === "2026-09"
   399	         ? configuredDailyAllowance(runtime, now)
   400	         : Math.min(remainingMonth, Math.ceil(remainingMonth / Math.max(1, remainingDays)));
   401	      const todayReservedRows = await tx.select({
   402	        sourceType: outreachMessagesTable.sourceType,
   403	        contactEmail: prospectsTable.contactEmail,
   404	        contactName: prospectsTable.contactName,
   405	        contactEvidenceType: prospectsTable.contactEvidenceType,
   406	      }).from(outreachMonthlySendReservationsTable)
   407	        .innerJoin(outreachMessagesTable, eq(outreachMonthlySendReservationsTable.messageId, outreachMessagesTable.id))
   408	        .innerJoin(prospectsTable, eq(outreachMessagesTable.prospectId, prospectsTable.id))
   409	        .where(and(eq(outreachMonthlySendReservationsTable.quotaKey, monthlyQuotaKey), gte(outreachMonthlySendReservationsTable.createdAt, dayStart), inArray(outreachMessagesTable.status, ["approved", "sending", "needs_review"])));
   410	       const todayReserved = todayReservedRows.length;
   411	      const uncappedHotLead = lane === "hot_lead" && effectiveAllocations?.hotLead === OUTREACH_UNCAPPED;
   412	       if (remainingMonth <= 0 || (!uncappedHotLead && todayReserved >= dailyAllowance)) {
   413	        throw new DailySendLimitError();
   414	      }
   415	      const campaignCap = campaign?.dailyLimit;
   416	      if (monthKey !== "2026-09" && campaign?.id && campaignCap && campaignCap > 0) {
   417	        const [campaignUsedToday] = await tx.select({ value: count() })
   418	          .from(outreachMessagesTable)
   419	          .where(and(
   420	            eq(outreachMessagesTable.campaignId, campaign.id),
   421	            eq(outreachMessagesTable.sequenceNumber, 1),
   422	            gte(outreachMessagesTable.sentAt, dayStart),
   423	          ));
   424	        const [campaignReservedToday] = await tx.select({ value: count() })
   425	          .from(outreachMonthlySendReservationsTable)
   426	          .innerJoin(
   427	            outreachMessagesTable,
   428	            eq(outreachMonthlySendReservationsTable.messageId, outreachMessagesTable.id),
   429	          )
   430	          .where(and(
   431	            eq(outreachMessagesTable.campaignId, campaign.id),
   432	            gte(outreachMonthlySendReservationsTable.createdAt, dayStart),
   433	            inArray(outreachMessagesTable.status, ["approved", "sending", "needs_review"]),
   434	          ));
   435	        if ((campaignUsedToday?.value ?? 0) + (campaignReservedToday?.value ?? 0) >= campaignCap) {
   436	          throw new DailySendLimitError();
   437	        }
   438	      }
   439	      if (monthKey === "2026-09") {
   440	        const [laneRows, laneReservations] = await Promise.all([
   441	          tx.select({
   442	            sourceType: outreachMessagesTable.sourceType,
   443	            contactEmail: prospectsTable.contactEmail,
   444	            contactName: prospectsTable.contactName,
   445	            contactEvidenceType: prospectsTable.contactEvidenceType,
   446	          }).from(outreachMessagesTable)
   447	            .innerJoin(prospectsTable, eq(outreachMessagesTable.prospectId, prospectsTable.id))
   448	            .where(and(eq(outreachMessagesTable.sequenceNumber, 1), gte(outreachMessagesTable.sentAt, dayStart))),
   449	          tx.select({
   450	            sourceType: outreachMessagesTable.sourceType,
   451	            contactEmail: prospectsTable.contactEmail,
   452	            contactName: prospectsTable.contactName,
   453	            contactEvidenceType: prospectsTable.contactEvidenceType,
   454	          }).from(outreachMonthlySendReservationsTable)
   455	            .innerJoin(outreachMessagesTable, eq(outreachMonthlySendReservationsTable.messageId, outreachMessagesTable.id))
   456	            .innerJoin(prospectsTable, eq(outreachMessagesTable.prospectId, prospectsTable.id))
   457	            .where(and(eq(outreachMonthlySendReservationsTable.quotaKey, monthlyQuotaKey), gte(outreachMonthlySendReservationsTable.createdAt, dayStart), inArray(outreachMessagesTable.status, ["approved", "sending", "needs_review"]))),
   458	        ]);
   459	        const laneCount = [...laneRows, ...laneReservations]
   460	          .filter((row) => getOutreachDailyLane(row, row) === lane).length;
   461	        const laneLimit = effectiveAllocations
   462	          ? getOutreachDailyLaneLimit(lane, {
   463	            namedLimit: effectiveAllocations.named,
   464	            publicLimit: effectiveAllocations.public,
   465	            hotMarketLimit: effectiveAllocations.hotMarket,
   466	            hotLeadLimit: effectiveAllocations.hotLead,
   467	          })
   468	          : undefined;
   469	        if (laneLimit !== undefined && !isUncappedLaneLimit(laneLimit)) {
   470	           const sharedLane = lane === "named" || lane === "direct"
   471	             || lane === "public" || lane === "hot_market";
   472	          const sharedLimit = effectiveAllocations?.namedHotMarketShared;
   473	          if (sharedLane && sharedLimit !== undefined) {
   474	            const sharedCount = [...laneRows, ...laneReservations]
   475	              .filter((row) => {
   476	                const rowLane = getOutreachDailyLane(row, row);
   477	                 return rowLane === "named" || rowLane === "public" || rowLane === "hot_market";
   478	              }).length;
   479	            if (sharedCount >= sharedLimit) throw new DailySendLimitError();
   480	          } else if (laneCount >= laneLimit) {
   481	            throw new DailySendLimitError();
   482	          }
   483	        }
   484	      }
   485	      for (let slot = used + 1; slot <= monthlyTarget; slot += 1) {
   486	        const [inserted] = await tx.insert(outreachMonthlySendReservationsTable)
   487	          .values({
   488	            messageId: message.id,
   489	            normalizedEmail,
   490	            sequenceNumber: message.sequenceNumber,
   491	            quotaKey: monthlyQuotaKey,
   492	            slot,
   493	          })
   494	          .onConflictDoNothing()
   495	          .returning({ id: outreachMonthlySendReservationsTable.id });
   496	        if (inserted) {
   497	          dailyReservationId = inserted.id;
   498	          break;
   499	        }
   500	      }
   501	      if (!dailyReservationId) throw new MonthlySendLimitError();
   502	    }
   503	
   504	    return {
   505	      dailyReservationId,
   506	      regularMonthlyReservationId: dailyReservationId,
   507	      sequenceClaimId: sequenceClaim.id,
   508	    };
   509	  });
   510	}

   585	function isDraft(value: unknown): value is Pick<GeneratedDraft, "subject" | "body"> {
   586	  if (!value || typeof value !== "object") return false;

   588	  return typeof draft.subject === "string" && typeof draft.body === "string" &&
   589	    !("followUps" in draft);

   592	export async function sendApprovedOutreach(
   593	  message: OutreachMessage,
   594	  prospect: Prospect,
   595	  campaign: Campaign | undefined,
   596	  options: OutreachSendOptions = {},
   597	): Promise<{ providerMessageId?: string }> {
   598	  const [currentProspect] = await db.select().from(prospectsTable).where(eq(prospectsTable.id, prospect.id));
   599	  if (!currentProspect) throw new Error("Prospect not found");
   600	  const [currentCampaign] = campaign
   601	    ? await db.select().from(campaignsTable).where(eq(campaignsTable.id, campaign.id))
   602	    : [];
   603	  if (campaign && !currentCampaign) throw new Error("Campaign not found");
   604	  const email = assertOutreachEligibilityBase(message, currentProspect, currentCampaign);
   605	  assertScheduledTimeReady(message.sequenceNumber, message.scheduledAt);
   606	  if (message.sequenceNumber > 1) {
   607	    const initialDeliveredAt = await getVerifiedInitialDeliveryAt(message);
   608	    const initialEngagedAt = await getVerifiedInitialEngagementAt(message);
   609	    if (!initialDeliveredAt) {
   610	      throw new Error("Initial sequence message must have verified delivery evidence before this follow-up can send");
   611	    }
   612	    assertFollowUpCadenceReady(
   613	      message.sequenceNumber,
   614	      message.scheduledAt,
   615	      initialEngagedAt,
   616	    );
   617	    const campaignScope = message.campaignId
   618	      ? eq(outreachMessagesTable.campaignId, message.campaignId)
   619	      : isNull(outreachMessagesTable.campaignId);
   620	    const [previous] = await db.select({ status: outreachMessagesTable.status })
   621	      .from(outreachMessagesTable)
   622	      .where(and(
   623	        eq(outreachMessagesTable.prospectId, message.prospectId),
   624	        campaignScope,
   625	        eq(outreachMessagesTable.sequenceNumber, message.sequenceNumber - 1),
   626	      ))
   627	      .limit(1);
   628	    assertSequenceDeliveryReady(message.sequenceNumber, previous?.status);
   629	  }
   630	  const [suppression] = await db.select({ id: outreachSuppressionsTable.id }).from(outreachSuppressionsTable).where(eq(outreachSuppressionsTable.email, email)).limit(1);
   631	  if (suppression) throw new Error("Address is suppressed");
   632	
   633	  const unsubscribeUrl = options.unsubscribeUrls?.unsubscribeUrl ?? makeUnsubscribeUrl(email);
   634	  const oneClickUrl = options.unsubscribeUrls?.oneClickUnsubscribeUrl ?? makeOneClickUnsubscribeUrl(email);
   635	  if (!unsubscribeUrl || !oneClickUrl) throw new Error("Unsubscribe signing is not configured");
   636	  const isHotMarketMessage = isHotMarketSourceType(message.sourceType);
   637	  const usesCurrentSharedCopy = !isHotMarketMessage
   638	    && currentCampaign?.bodyTemplate === "Approved personalized Apex Grid outreach copy";
   639	  const currentRegularFollowUp = !isHotMarketMessage && message.sequenceNumber === 2
   640	    ? approvedOutreachFollowUpMessages(currentProspect.contactName ?? "")
   641	      .find((followUp) => followUp.sequenceNumber === message.sequenceNumber)
   642	    : undefined;
   643	  const currentSubject = currentRegularFollowUp?.subject ?? (usesCurrentSharedCopy
   644	      ? approvedOutreachSubject(currentProspect.companyName)
   645	    : message.subject);
   646	  const currentBody = currentRegularFollowUp?.body ?? (usesCurrentSharedCopy
   647	      ? message.sequenceNumber === 1
   648	        ? approvedOutreachBody(currentProspect.contactName ?? "")
   649	        : message.body
   650	    : message.body);
   651	  const emailContent = renderBrandedEmail(currentBody, unsubscribeUrl, currentProspect.contactName ?? undefined);
   652	  const from = options.fromEmail?.trim() || process.env.OUTREACH_FROM_EMAIL;
   653	  if (!from) throw new Error("OUTREACH_FROM_EMAIL is not configured");
   654	  const replyTo = options.replyToEmail?.trim() || process.env.OUTREACH_REPLY_TO_EMAIL?.trim() || from;
   655	  const dedicatedSendGridKey = process.env.SENDGRID_ISOLATION_VERIFIED === "true"
   656	    ? process.env.SENDGRID_DEDICATED_API_KEY?.trim()
   657	    : undefined;
   658	  const sendgridSubuser = process.env.SENDGRID_SUBUSER_USERNAME?.trim();
   659	  const providerReconciliationKey = await ensureProviderReconciliationKey(message.id);
   660	  // A supplied dispatch is the test/injected provider boundary; production
   661	  // dispatches always come through the worker's explicit readiness result.
   662	  const readiness = options.deliverabilityReady ?? (options.dispatch ? true : undefined);
   663	  const reservations = await reserveOutreachSend(message, currentCampaign, email, readiness);
   664	  await options.beforeEmailLock?.();
   665	  return withOutreachEmailLock(email, async () => {
   666	    if (message.sequenceNumber === 2) {
   667	      const [lockedDelivery, lockedEngagement] = await Promise.all([
   668	        getVerifiedInitialDeliveryAt(message),
   669	        getVerifiedInitialEngagementAt(message),
   670	      ]);
   671	      if (!lockedDelivery || !lockedEngagement) {
   672	        throw new Error("Follow-up lost its verified delivery or engagement evidence before provider dispatch");
   673	      }
   674	      assertFollowUpCadenceReady(message.sequenceNumber, message.scheduledAt, lockedEngagement);
   675	    }
   676	    const [[lockedMessage], [lockedProspect], [lockedSuppression]] = await Promise.all([
   677	      db.select({ status: outreachMessagesTable.status })
   678	        .from(outreachMessagesTable)
   679	        .where(eq(outreachMessagesTable.id, message.id))
   680	        .limit(1),
   681	      db.select({
   682	        status: prospectsTable.status,
   683	        contactStatus: prospectsTable.contactStatus,
   684	        contactEmail: prospectsTable.contactEmail,
   685	      })
   686	        .from(prospectsTable)
   687	        .where(eq(prospectsTable.id, message.prospectId))
   688	        .limit(1),
   689	      db.select({ id: outreachSuppressionsTable.id })
   690	        .from(outreachSuppressionsTable)
   691	        .where(eq(outreachSuppressionsTable.email, email))
   692	        .limit(1),
   693	    ]);
   694	    const persistedStatusIsEligible = options.expectedPersistedStatus
   695	      ? lockedMessage?.status === options.expectedPersistedStatus
   696	      : lockedMessage?.status === "approved" || lockedMessage?.status === "sending";
   697	    const prospectIsEligible = lockedProspect?.contactStatus === "active"
   698	      && (lockedProspect.status === "approved" || lockedProspect.status === "contacted")
   699	      && lockedProspect.contactEmail?.trim().toLowerCase() === email;
   700	    if (!persistedStatusIsEligible || !prospectIsEligible || lockedSuppression) {
   701	      await releaseOutreachReservations(reservations);
   702	      throw new Error("Outreach stopped before provider dispatch");
   703	    }
   704	
   705	    let response: Awaited<ReturnType<ReplitConnectors["proxy"]>> | undefined;
   706	    const requestBody = JSON.stringify({
   707	      personalizations: [{
   708	        to: [{ email }],
   709	        headers: {
   710	          "List-Unsubscribe": `<${oneClickUrl}>`,
   711	          "List-Unsubscribe-Post": "List-Unsubscribe=One-Click",
   712	        },
   713	        custom_args: {
   714	          outreach_message_id: String(message.id),
   715	          outreach_prospect_id: String(prospect.id),
   716	          outreach_reconciliation_key: providerReconciliationKey,
   717	        },
   718	      }],
   719	      from: { email: from, name: "Apex Grid Engineering" },
   720	      reply_to: { email: replyTo, name: "Apex Grid Engineering" },
   721	      subject: currentSubject,
   722	      content: [
   723	        { type: "text/plain", value: emailContent.plainText },
   724	        { type: "text/html", value: emailContent.html },
   725	      ],
   726	    });
   727	    for (let attempt = 0; attempt < 4; attempt += 1) {
   728	      try {
   729	        await options.beforeProviderDispatch?.();
   730	        response = options.dispatch
   731	          ? await options.dispatch({ body: requestBody, email, fromEmail: from, replyToEmail: replyTo })
   732	          : dedicatedSendGridKey
   733	            ? await fetch("https://api.sendgrid.com/v3/mail/send", {
   734	              method: "POST",
   735	              headers: {
   736	                "Content-Type": "application/json",
   737	                Authorization: `Bearer ${dedicatedSendGridKey}`,
   738	              },
   739	              body: requestBody,
   740	            })
   741	            : await new ReplitConnectors().proxy("sendgrid", "/v3/mail/send", {
   742	              method: "POST",
   743	              headers: {
   744	                "Content-Type": "application/json",
   745	                ...(sendgridSubuser ? { "on-behalf-of": sendgridSubuser } : {}),
   746	              },
   747	              body: requestBody,
   748	            });
   749	      } catch {
   750	        throw new Error("SendGrid dispatch result is unknown; message requires reconciliation before retry");
   751	      }
   752	      if (response.ok || response.status !== 429 || attempt === 3) break;
   753	      // A response proves the provider did not accept this attempt. Pace the
   754	      // retry instead of consuming the daily lane with a false failure.
   755	      await new Promise((resolve) => setTimeout(resolve, 100 * 2 ** attempt));
   756	    }
   757	    if (!response) {
   758	      throw new Error("SendGrid dispatch result is unknown; message requires reconciliation before retry");
   759	    }
   760	    if (!response.ok) {
   761	    if (isDefinitiveSendGridRejection(response.status)) {
   762	      await releaseOutreachReservations(reservations);
   763	      throw new Error(`SendGrid rejected the message with status ${response.status}`);
   764	    }
   765	    if (response.status === 429) {
   766	      await releaseOutreachReservations(reservations);
   767	      throw new ProviderRateLimitError();
   768	    }
   769	    throw new Error(`SendGrid dispatch result is unknown after status ${response.status}; message requires reconciliation before retry`);
   770	    }
   771	    const providerMessageId = response.headers.get("x-message-id") ?? undefined;
   772	    await options.afterProviderDispatch?.(providerMessageId);
   773	    return { providerMessageId };
   774	  });
   775	}
