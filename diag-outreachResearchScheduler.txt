    29	export const OUTREACH_RESEARCH_TIMEZONE = "America/Phoenix";
    30	export const OUTREACH_RESEARCH_LOCAL_HOUR = 20;
    31	export const HOT_MARKET_RESEARCH_STATES_PER_RUN = 8;
    32	const HOT_MARKET_RESEARCH_QUERY_PREFIX = "hot-market-replenishment:";
    33	const HOT_MARKET_RESEARCH_AUDIENCES: ResearchAudience[] = ["builder", "architect"];
    34	
    35	const DATE_FORMATTER = new Intl.DateTimeFormat("en-CA", {
    36	  timeZone: OUTREACH_RESEARCH_TIMEZONE,
    37	  year: "numeric",
    38	  month: "2-digit",
    39	  day: "2-digit",
    40	  hour: "2-digit",
    41	  hourCycle: "h23",
    42	});
    43	
    44	export function getPhoenixResearchWindow(now: Date): { runDate: string; localHour: number } {
    45	  const parts = Object.fromEntries(
    46	    DATE_FORMATTER.formatToParts(now)
    47	      .filter((part) => part.type !== "literal")
    48	      .map((part) => [part.type, part.value]),
    49	  );
    50	  return {
    51	    runDate: `${parts.year}-${parts.month}-${parts.day}`,
    52	    localHour: Number(parts.hour),
    53	  };
    54	}

    56	export function isResearchScheduleDue(
    57	  schedule: Pick<ResearchSchedule, "enabled" | "localHour">,
    58	  campaign: Pick<Campaign, "status">,
    59	  now: Date,
    60	): boolean {
    61	  if (!schedule.enabled || campaign.status !== "active") return false;
    62	  return getPhoenixResearchWindow(now).localHour >= OUTREACH_RESEARCH_LOCAL_HOUR;
    63	}

    65	export function usesGenericResearchPipeline(campaign: Pick<Campaign, "name">): boolean {
    66	  return !isRecurringHotMarketCampaign(campaign);
    67	}

    69	export function getDailyResearchTarget(scheduleTarget: number, campaignLimit: number): number {
    70	  return Math.min(
    71	    getOutreachDiscoveryDailyCap(),
    72	    Math.max(1, scheduleTarget, campaignLimit),
    73	  );
    74	}

    76	function validStates(states: string[]): ResearchState[] {
    77	  return states.filter((state): state is ResearchState =>
    78	    (RESEARCH_STATE_ORDER as readonly string[]).includes(state)
    79	  );
    80	}
    81	
    82	function validAudiences(audience: string): ResearchAudience[] {
    83	  if (audience === "mixed") return ["architect", "builder"];
    84	  if (audience === "architect" || audience === "builder") return [audience];
    85	  throw new Error(`Unsupported campaign audience: ${audience}`);
    86	}

    88	function uniqueCandidates(candidates: DiscoveredProspect[]): DiscoveredProspect[] {
    89	  const seen = new Set<string>();
    90	  return candidates.filter((candidate) => {
    91	    if (seen.has(candidate.dedupeKey)) return false;
    92	    seen.add(candidate.dedupeKey);
    93	    return true;
    94	  });
    95	}

    97	export function getHotMarketResearchStates(
    98	  runDate: string,
    99	  limit = HOT_MARKET_RESEARCH_STATES_PER_RUN,
   100	): ResearchState[] {
   101	  const secondaryStates = RESEARCH_STATE_ORDER.slice(2);
   102	  const dayNumber = Math.floor(new Date(`${runDate}T12:00:00Z`).getTime() / 86_400_000);
   103	  const secondaryCount = Math.max(0, Math.min(secondaryStates.length, limit - 2));
   104	  const offset = secondaryStates.length === 0 ? 0 : dayNumber % secondaryStates.length;
   105	  const rotatingStates = Array.from({ length: secondaryCount }, (_, index) =>
   106	    secondaryStates[(offset + index) % secondaryStates.length]
   107	  ).filter((state): state is ResearchState => Boolean(state));
   108	  return [...RESEARCH_STATE_ORDER.slice(0, Math.min(2, limit)), ...rotatingStates];
   109	}

   111	export function getHotMarketResearchTarget(
   112	  now = new Date(),
   113	): { targetDate: string; scheduledAt: Date } {
   114	  const { runDate } = getPhoenixResearchWindow(now);
   115	  const scheduledAt = new Date(
   116	    new Date(`${runDate}T20:10:00-07:00`).getTime() + 24 * 60 * 60 * 1000,
   117	  );
   118	  return {
   119	    targetDate: getPhoenixResearchWindow(scheduledAt).runDate,
   120	    scheduledAt,
   121	  };
   122	}

   124	function uniqueHotMarketCandidates(
   125	  candidates: DiscoveredHotMarketProspect[],
   126	): DiscoveredHotMarketProspect[] {
   127	  const dedupeKeys = new Set<string>();
   128	  const emails = new Set<string>();
   129	  const domains = new Set<string>();
   130	  return candidates.filter((candidate) => {
   131	    const email = candidate.contactEmail.trim().toLowerCase();
   132	    let domain = "";
   133	    try {
   134	      domain = new URL(candidate.website).hostname.replace(/^www\./, "").toLowerCase();
   135	    } catch {
   136	      return false;
   137	    }
   138	    if (
   139	      dedupeKeys.has(candidate.dedupeKey)
   140	      || emails.has(email)
   141	      || domains.has(domain)
   142	    ) return false;
   143	    dedupeKeys.add(candidate.dedupeKey);
   144	    emails.add(email);
   145	    domains.add(domain);
   146	    return true;
   147	  });
   148	}

   150	async function claimHotMarketResearchRun(
   151	  runDate: string,
   152	  states: ResearchState[],
   153	  now: Date,
   154	): Promise<number | undefined> {
   155	  const query = `${HOT_MARKET_RESEARCH_QUERY_PREFIX}${runDate}`;
   156	  return db.transaction(async (tx) => {
   157	    await tx.execute(sql`select pg_advisory_xact_lock(hashtextextended(${query}, 0))`);
   158	    const [existing] = await tx.select().from(outreachResearchRunsTable)
   159	      .where(eq(outreachResearchRunsTable.query, query))
   160	      .limit(1);
   161	    if (existing) {
   162	      const stale = existing.status === "running"
   163	        && existing.createdAt.getTime() <= now.getTime() - 30 * 60_000;
   164	      if (existing.status !== "failed" && !stale) return undefined;
   165	      const [recovered] = await tx.update(outreachResearchRunsTable).set({
   166	        state: states.join(","),
   167	        status: "running",
   168	        resultCount: 0,
   169	        skippedCount: 0,
   170	        error: null,
   171	        completedAt: null,
   172	      }).where(eq(outreachResearchRunsTable.id, existing.id)).returning({
   173	        id: outreachResearchRunsTable.id,
   174	      });
   175	      return recovered?.id;
   176	    }
   177	    const [created] = await tx.insert(outreachResearchRunsTable).values({
   178	      state: states.join(","),
   179	      audience: "mixed",
   180	      query,
   181	      status: "running",
   182	    }).returning({ id: outreachResearchRunsTable.id });
   183	    return created?.id;
   184	  });
   185	}

   187	export async function processDueHotMarketResearch(
   188	  now = new Date(),
   189	): Promise<{ state: "skipped" | "completed" | "failed"; staged: number; shortfall: number }> {
   190	  const { runDate, localHour } = getPhoenixResearchWindow(now);
   191	  if (localHour < OUTREACH_RESEARCH_LOCAL_HOUR) {
   192	    return { state: "skipped", staged: 0, shortfall: HOT_MARKET_DAILY_TARGET };
   193	  }
   194	  const states = getHotMarketResearchStates(runDate);
   195	  const runId = await claimHotMarketResearchRun(runDate, states, now);
   196	  if (!runId) return { state: "skipped", staged: 0, shortfall: 0 };
   197	
   198	  try {
   199	    const discoveries: DiscoveredHotMarketProspect[] = [];
   200	    for (const state of states) {
   201	      for (const audience of HOT_MARKET_RESEARCH_AUDIENCES) {
   202	        const result = await discoverPublicHotMarketProspects({ state, audience });
   203	        discoveries.push(...result.prospects);
   204	      }
   205	    }
   206	    const candidates = uniqueHotMarketCandidates(discoveries);
   207	    const target = getHotMarketResearchTarget(now);
   208	    const result = await stageVerifiedHotMarketProspects(candidates, {
   209	      targetDate: target.targetDate,
   210	      scheduledAt: target.scheduledAt,
   211	      targetCount: HOT_MARKET_DAILY_TARGET,
   212	      now,
   213	    });
   214	    await db.update(outreachResearchRunsTable).set({
   215	      status: "completed",
   216	      resultCount: result.staged,
   217	      skippedCount: result.skipped,
   218	      completedAt: new Date(),
   219	      error: result.shortfall > 0
   220	        ? `Verified hot-market shortfall: ${result.shortfall}`
   221	        : null,
   222	    }).where(eq(outreachResearchRunsTable.id, runId));
   223	    return { state: "completed", staged: result.staged, shortfall: result.shortfall };
   224	  } catch (error) {
   225	    const message = error instanceof Error ? error.message : "Hot-market research failed";
   226	    await db.update(outreachResearchRunsTable).set({
   227	      status: "failed",
   228	      error: message,
   229	      completedAt: new Date(),
   230	    }).where(eq(outreachResearchRunsTable.id, runId));
   231	    logger.error({ err: error, runDate }, "Scheduled hot-market research run failed");
   232	    return { state: "failed", staged: 0, shortfall: HOT_MARKET_DAILY_TARGET };
   233	  }
   234	}

   236	async function completeScheduledResearch(
   237	  schedule: ResearchSchedule,
   238	  campaign: Campaign,
   239	  scheduleRunId: number,
   240	): Promise<void> {
   241	  const states = validStates(campaign.states);
   242	  if (states.length === 0) throw new Error("Campaign has no supported target states");
   243	
   244	  const audiences = validAudiences(campaign.audience);
   245	  const [researchRun] = await db.insert(outreachResearchRunsTable).values({
   246	    campaignId: campaign.id,
   247	    state: states.join(","),
   248	    audience: campaign.audience,
   249	    query: `Scheduled research for ${campaign.name}`,
   250	    status: "running",
   251	  }).returning();
   252	  if (!researchRun) throw new Error("Unable to create research run");
   253	
   254	  try {
   255	    const discoveries = [];
   256	    for (const state of states) {
   257	      for (const audience of audiences) {
   258	        discoveries.push(await discoverPublicProspects({ state, audience }));
   259	      }
   260	    }
   261	
   262	    const queries = discoveries.map((discovery) => discovery.query);
   263	    const candidates = uniqueCandidates(discoveries.flatMap((discovery) => discovery.prospects));
   264	    const existingRows = candidates.length === 0
   265	      ? []
   266	      : await db.select({ dedupeKey: prospectsTable.dedupeKey })
   267	        .from(prospectsTable)
   268	        .where(inArray(prospectsTable.dedupeKey, candidates.map((candidate) => candidate.dedupeKey)));
   269	    const existing = new Set(existingRows.map((row) => row.dedupeKey));
   270	    const targetCount = getDailyResearchTarget(schedule.targetCount, campaign.dailyLimit);
   271	    const newCandidates = candidates
   272	      .filter((candidate) => !existing.has(candidate.dedupeKey))
   273	      .slice(0, targetCount);
   274	    const inserted = newCandidates.length === 0
   275	      ? []
   276	      : await db.insert(prospectsTable).values(newCandidates.map((candidate) => ({
   277	        ...candidate,
   278	        campaignId: campaign.id,
   279	        researchRunId: researchRun.id,
   280	        status: "review",
   281	        emailStatus: "unverified",
   282	        contactConfidence: "unknown",
   283	      }))).onConflictDoNothing().returning();
   284	    const skippedCount = Math.max(0, candidates.length - inserted.length);
   285	    const completedAt = new Date();
   286	    const query = queries.join(" | ");
   287	
   288	    await Promise.all([
   289	      db.update(outreachResearchRunsTable).set({
   290	        query,
   291	        status: "completed",
   292	        resultCount: inserted.length,
   293	        skippedCount,
   294	        completedAt,
   295	      }).where(eq(outreachResearchRunsTable.id, researchRun.id)),
   296	      db.update(outreachResearchScheduleRunsTable).set({
   297	        query,
   298	        status: "completed",
   299	        resultCount: inserted.length,
   300	        skippedCount,
   301	        completedAt,
   302	      }).where(eq(outreachResearchScheduleRunsTable.id, scheduleRunId)),
   303	    ]);
   304	  } catch (error) {
   305	    const message = error instanceof Error ? error.message : "Scheduled research failed";
   306	    const completedAt = new Date();
   307	    await Promise.all([
   308	      db.update(outreachResearchRunsTable).set({
   309	        status: "failed",
   310	        error: message,
   311	        completedAt,
   312	      }).where(eq(outreachResearchRunsTable.id, researchRun.id)),
   313	      db.update(outreachResearchScheduleRunsTable).set({
   314	        status: "failed",
   315	        error: message,
   316	        completedAt,
   317	      }).where(eq(outreachResearchScheduleRunsTable.id, scheduleRunId)),
   318	    ]);
   319	    throw error;
   320	  }
   321	}

   323	export async function processDueOutreachResearchSchedules(now = new Date()): Promise<number> {
   324	  const rows = await db.select({
   325	    schedule: outreachResearchSchedulesTable,
   326	    campaign: campaignsTable,
   327	  })
   328	    .from(outreachResearchSchedulesTable)
   329	    .innerJoin(campaignsTable, eq(outreachResearchSchedulesTable.campaignId, campaignsTable.id))
   330	    .where(and(
   331	      eq(outreachResearchSchedulesTable.enabled, true),
   332	      eq(campaignsTable.status, "active"),
   333	    ));
   334	  const { runDate } = getPhoenixResearchWindow(now);
   335	  let completed = 0;
   336	
   337	  for (const { schedule, campaign } of rows) {
   338	    if (!usesGenericResearchPipeline(campaign)) continue;
   339	    if (!isResearchScheduleDue(schedule, campaign, now)) continue;
   340	    const staleCutoff = new Date(now.getTime() - 30 * 60_000);
   341	    const [recovered] = await db.update(outreachResearchScheduleRunsTable).set({
   342	      status: "running",
   343	      startedAt: now,
   344	      error: null,
   345	      completedAt: null,
   346	    }).where(and(
   347	      eq(outreachResearchScheduleRunsTable.scheduleId, schedule.id),
   348	      eq(outreachResearchScheduleRunsTable.runDate, runDate),
   349	      eq(outreachResearchScheduleRunsTable.status, "running"),
   350	      lte(outreachResearchScheduleRunsTable.startedAt, staleCutoff),
   351	    )).returning();
   352	    const [claimed] = recovered
   353	      ? [recovered]
   354	      : await db.insert(outreachResearchScheduleRunsTable).values({
   355	          scheduleId: schedule.id,
   356	          runDate,
   357	          status: "running",
   358	        }).onConflictDoNothing().returning();
   359	    if (!claimed) continue;
   360	
   361	    try {
   362	      await completeScheduledResearch(schedule, campaign, claimed.id);
   363	      completed += 1;
   364	    } catch (error) {
   365	      logger.error({ err: error, scheduleId: schedule.id }, "Scheduled outreach research run failed");
   366	    }
   367	  }
   368	  return completed;
   369	}
