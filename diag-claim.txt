   139	export async function claimOutreachMessageForSending(messageId: number): Promise<OutreachMessage | undefined> {
   140	  const [claimed] = await db.update(outreachMessagesTable)
   141	    .set({ status: "sending", error: null })
   142	    .where(and(
   143	      eq(outreachMessagesTable.id, messageId),
   144	      eq(outreachMessagesTable.status, "approved"),
   145	      sql`exists (
   146	        select 1
   147	        from ${prospectsTable} as current_prospect
   148	        where current_prospect.id = ${outreachMessagesTable.prospectId}
   149	          and current_prospect.contact_status = 'active'
   150	          and current_prospect.status in ('approved', 'contacted')
   151	          and not exists (
   152	            select 1
   153	            from outreach_suppressions as suppression
   154	            where suppression.email = lower(trim(current_prospect.contact_email))
   155	          )
   156	      )`,
   157	      or(
   158	        eq(outreachMessagesTable.sequenceNumber, 1),
   159	        sql`exists (
   160	          select 1
   161	          from ${outreachMessagesTable} as previous_message
   162	          where previous_message.prospect_id = ${outreachMessagesTable.prospectId}
   163	            and previous_message.sequence_number = ${outreachMessagesTable.sequenceNumber} - 1
   164	            and previous_message.status = 'delivered'
   165	            and (
   166	              previous_message.campaign_id = ${outreachMessagesTable.campaignId}
   167	              or (
   168	                previous_message.campaign_id is null
   169	                and ${outreachMessagesTable.campaignId} is null
   170	              )
   171	            )
   172	        )`,
   173	      ),
   174	      sql`${outreachMessagesTable.catchUpCohortId} is null or exists (
   175	        select 1 from outreach_catch_up_reservations as catch_up_reservation
   176	        where catch_up_reservation.message_id = ${outreachMessagesTable.id}
   177	          and catch_up_reservation.status = 'reserved'
   178	      )`,
   179	      or(
   180	        eq(outreachMessagesTable.sequenceNumber, 1),
   181	        sql`exists (
   182	          select 1
   183	          from ${outreachMessagesTable} as delivered_initial
   184	          inner join outreach_delivery_events as delivery_evidence
   185	            on delivery_evidence.outreach_message_id = delivered_initial.id
   186	          where delivered_initial.prospect_id = ${outreachMessagesTable.prospectId}
   187	            and delivered_initial.sequence_number = 1
   188	            and delivered_initial.status = 'delivered'
   189	            and delivery_evidence.event_type = 'delivered'
   190	            and (
   191	              delivered_initial.campaign_id = ${outreachMessagesTable.campaignId}
   192	              or (
   193	                delivered_initial.campaign_id is null
   194	                and ${outreachMessagesTable.campaignId} is null
   195	              )
   196	            )
   197	        )`,
   198	      ),
   199	      sql`${outreachMessagesTable.sequenceNumber} <= 2`,
   200	      or(
   201	        eq(outreachMessagesTable.sequenceNumber, 1),
   202	        sql`exists (
   203	          select 1
   204	          from ${outreachMessagesTable} as initial_message
   205	          inner join outreach_delivery_events as initial_engagement
   206	            on initial_engagement.outreach_message_id = initial_message.id
   207	          where initial_message.prospect_id = ${outreachMessagesTable.prospectId}
   208	            and initial_message.sequence_number = 1
   209	            and initial_engagement.event_type in ('open', 'click')
   210	             and initial_engagement.occurred_at >= now() - (${OPENER_FOLLOW_UP_MAX_AGE_DAYS} * interval '1 day')
   211	              and initial_engagement.occurred_at <= now()
   212	            and exists (
   213	              select 1
   214	              from outreach_delivery_events as delivery_evidence
   215	              where delivery_evidence.outreach_message_id = initial_message.id
   216	                and delivery_evidence.event_type = 'delivered'
   217	                and delivery_evidence.occurred_at <= initial_engagement.occurred_at
   218	            )
   219	            and (
   220	              initial_message.campaign_id = ${outreachMessagesTable.campaignId}
   221	              or (
   222	                initial_message.campaign_id is null
   223	                and ${outreachMessagesTable.campaignId} is null
   224	              )
   225	            )
   226	        )`,
   227	      ),
   228	    ))
   229	    .returning();
   230	  if (!claimed) {
   231	    // Only an opener still approved can have been rejected by eligibility
   232	    // checks; a concurrent claimant is already sending and keeps its slot.
   233	    await db.update(outreachCatchUpReservationsTable).set({
   234	      status: "released",
   235	      releasedAt: new Date(),
   236	    }).where(and(
   237	      eq(outreachCatchUpReservationsTable.messageId, messageId),
   238	      eq(outreachCatchUpReservationsTable.status, "reserved"),
   239	      sql`exists (
   240	        select 1 from outreach_messages rejected
   241	        where rejected.id = ${messageId} and rejected.status = 'approved'
   242	      )`,
   243	    ));
   244	  }
   245	  return claimed;
   246	}

   248	export async function releaseDefinitiveCatchUpReservation(messageId: number): Promise<void> {
   249	  await db.update(outreachCatchUpReservationsTable).set({
   250	    status: "released",
   251	    releasedAt: new Date(),
   252	  }).where(and(
   253	    eq(outreachCatchUpReservationsTable.messageId, messageId),
   254	    eq(outreachCatchUpReservationsTable.status, "reserved"),
   255	  ));
   256	}
