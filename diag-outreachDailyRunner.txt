     1	import { NIGHTLY_TOTAL_INITIAL_TARGET } from "./outreachSystemConfig";
     2	import { getNextPhoenixPreparationTarget } from "./outreachPreparation";
     3	const OUTREACH_RESEARCH_TIMEZONE = "America/Phoenix";
     4	const OUTREACH_RESEARCH_LOCAL_HOUR = 20;
     5	const PHOENIX_DISPATCH_MINUTE = 10;
     6	const PHOENIX_OFFSET = "-07:00";
     7	
     8	const phoenixTimeFormatter = new Intl.DateTimeFormat("en-CA", {
     9	  timeZone: OUTREACH_RESEARCH_TIMEZONE,
    10	  year: "numeric",
    11	  month: "2-digit",
    12	  day: "2-digit",
    13	  hour: "2-digit",
    14	  minute: "2-digit",
    15	  hourCycle: "h23",
    16	});
    17	
    18	type PhoenixDateTime = {
    19	  date: string;
    20	  hour: number;
    21	  minute: number;
    22	};
    23	
    24	function getPhoenixDateTime(now: Date): PhoenixDateTime {
    25	  const parts = Object.fromEntries(
    26	    phoenixTimeFormatter.formatToParts(now)
    27	      .filter((part) => part.type !== "literal")
    28	      .map((part) => [part.type, part.value]),
    29	  );
    30	  return {
    31	    date: `${parts.year}-${parts.month}-${parts.day}`,
    32	    hour: Number(parts.hour),
    33	    minute: Number(parts.minute),
    34	  };
    35	}
    36	
    37	export function isPrimaryPhoenixInvocation(now: Date): boolean {
    38	  const local = getPhoenixDateTime(now);
    39	  return local.hour === OUTREACH_RESEARCH_LOCAL_HOUR && local.minute === 0;
    40	}
    41	
    42	/**
    43	 * Scheduled Deployments start at 20:00 Phoenix. Only that invocation waits
    44	 * for messages staged by research for 20:10; a later/manual invocation is a
    45	 * catch-up run and must never be held open.
    46	 */
    47	export function getPhoenixStagedMessageWaitMs(invokedAt: Date, now: Date): number {
    48	  const invoked = getPhoenixDateTime(invokedAt);
    49	  if (
    50	    invoked.hour !== OUTREACH_RESEARCH_LOCAL_HOUR
    51	    || invoked.minute !== 0
    52	  ) return 0;
    53	
    54	  const dispatchAt = new Date(
    55	    `${invoked.date}T${String(OUTREACH_RESEARCH_LOCAL_HOUR).padStart(2, "0")}:${String(PHOENIX_DISPATCH_MINUTE).padStart(2, "0")}:00${PHOENIX_OFFSET}`,
    56	  );
    57	  return Math.max(0, dispatchAt.getTime() - now.getTime());
    58	}

   155	export function calculateVerifiedTopUpNeeded(windowCount: number): number {

   159	export type DailyOutreachLease = {
   160	  tryAcquire: () => Promise<boolean>;
   161	  release: () => Promise<void>;
   162	};
   163	
   164	export type ExclusiveDailyOutreachResult<T> =
   165	  | { state: "busy" }
   166	  | { state: "completed"; result: T };
   167	
   168	export async function withExclusiveDailyOutreachRun<T>(
   169	  lease: DailyOutreachLease,
   170	  task: () => Promise<T>,
   171	): Promise<ExclusiveDailyOutreachResult<T>> {
   172	  if (!await lease.tryAcquire()) return { state: "busy" };
   173	  try {
   174	    return { state: "completed", result: await task() };
   175	  } finally {
   176	    await lease.release();
   177	  }
   178	}

   180	/**
   181	 * Performs one bounded outreach pass. Each operation delegates to the
   182	 * production worker's database-claimed implementations, so concurrent
   183	 * autoscale workers cannot double-send a message.
   184	 */
   185	export async function runDailyOutreachOnce(
   186	  operations: DailyOutreachRunnerOperations,
   187	  invokedAt = operations.now(),
   188	): Promise<DailyOutreachRunnerResult> {
   189	  type VerificationSummary = {
   190	    promoted?: number;
   191	    finderCalls?: number;
   192	    creditBlocked?: boolean;
   193	  };
   194	  const researchOperations = [
   195	    ["hot-market-research", operations.processHotMarketResearch],
   196	    ["scheduled-research", operations.processScheduledResearch],
   197	    ["verification", operations.verifyProspects],
   198	  ] as const;
   199	  const acquisitionErrors: Array<{ stage: string; message: string }> = [];
   200	  const readSummary = (value: unknown): VerificationSummary => {
   201	    if (!value || typeof value !== "object") return {};
   202	    const summary = value as { promoted?: unknown; finderCalls?: unknown; creditBlocked?: unknown };
   203	    return {
   204	      promoted: typeof summary.promoted === "number" ? summary.promoted : undefined,
   205	      finderCalls: typeof summary.finderCalls === "number" ? summary.finderCalls : undefined,
   206	      creditBlocked: typeof summary.creditBlocked === "boolean" ? summary.creditBlocked : undefined,
   207	    };
   208	  };
   209	  let verificationSummary: VerificationSummary = {};
   210	  for (const [stage, operation] of researchOperations) {
   211	    if (!operation) continue;
   212	    try {
   213	      const outcome = await operation();
   214	      if (stage === "verification") verificationSummary = readSummary(outcome);
   215	    } catch (error) {
   216	      if (stage === "verification") {
   217	        // A credit-blocked verification still reports what it accomplished
   218	        // before stopping (the wire attaches the summary to the error).
   219	        verificationSummary = readSummary((error as { verificationSummary?: unknown } | null)?.verificationSummary);
   220	      }
   221	      // Acquisition failures are recorded on the result instead of being
   222	      // silently swallowed: a failed research/verification stage must show up
   223	      // in the daily report, not just in the logs.
   224	      acquisitionErrors.push({
   225	        stage,
   226	        message: error instanceof Error ? error.message : String(error),
   227	      });
   228	      operations.onResearchError?.(stage, error);
   229	    }
   230	  }
   231	  const regularPreparation = await operations.prepareRegularOutreach();
   232	  // Hot-market contacts are selected by regular preparation as part of the
   233	  // unified verified pool. The former dedicated lane must not stage extras.
   234	  const hotMarketPreparation = {
   235	    state: "skipped" as const,
   236	    prepared: 0,
   237	    totalScheduled: 0,
   238	    shortfall: 0,
   239	  };
   240	  const hotLeadPreparation = operations.prepareHotLeadOutreach
   241	    ? await operations.prepareHotLeadOutreach()
   242	    : { state: "skipped" as const, prepared: 0, totalScheduled: 0, shortfall: 100 };
   243	  const scheduledAt = getNextPhoenixPreparationTarget(invokedAt).scheduledAt;
   244	  const windowCount = operations.countInitialMessagesInWindow
   245	    ? await operations.countInitialMessagesInWindow(scheduledAt)
   246	    : regularPreparation.prepared + hotMarketPreparation.totalScheduled
   247	      + hotLeadPreparation.totalScheduled;
   248	  const topUpNeeded = calculateVerifiedTopUpNeeded(windowCount);
   249	  const topUp = operations.topUpVerifiedPreparation && topUpNeeded > 0
   250	    ? await operations.topUpVerifiedPreparation(scheduledAt, topUpNeeded)
   251	    : operations.topUpVerifiedPreparation
   252	      ? { prepared: 0, shortfall: 0 }
   253	      : undefined;
   254	
   255	  const initial = await operations.processDueMessages();
   256	  const reconciliation = operations.processProviderReconciliation();
   257	  const waitMs = getPhoenixStagedMessageWaitMs(invokedAt, operations.now());
   258	  if (waitMs === 0) {
   259	    const reconciled = await reconciliation;
   260	    return {
   261	      claimed: initial.claimed,
   262	      providerAccepted: initial.providerAccepted + reconciled.accepted,
   263	      delivered: 0,
   264	      bounced: reconciled.failed,
   265	      stopped: initial.stopped,
   266	      unresolved: initial.unresolved + reconciled.ambiguous,
   267	      regularPrepared: regularPreparation.prepared,
   268	      directPrepared: regularPreparation.directPrepared,
   269	      publicPrepared: regularPreparation.publicPrepared,
   270	      directShortfall: regularPreparation.directShortfall,
   271	      publicShortfall: regularPreparation.publicShortfall,
   272	      regularShortfall: regularPreparation.shortfall,
   273	      hotMarketPrepared: hotMarketPreparation.prepared,
   274	      hotMarketScheduled: hotMarketPreparation.totalScheduled,
   275	      hotMarketShortfall: hotMarketPreparation.shortfall,
   276	       hotLeadTarget: hotLeadPreparation.totalScheduled + hotLeadPreparation.shortfall,
   277	       hotLeadPrepared: hotLeadPreparation.totalScheduled,
   278	       hotLeadShortfall: hotLeadPreparation.shortfall,
   279	       ...(topUp ? { topUpPrepared: topUp.prepared, topUpShortfall: topUp.shortfall } : {}),
   280	       acquisitionErrors,
   281	       verificationPromoted: verificationSummary.promoted,
   282	       verificationFinderCalls: verificationSummary.finderCalls,
   283	       verificationCreditBlocked: verificationSummary.creditBlocked,
   284	      waitMs,
   285	    };
   286	  }
   287	
   288	  await operations.wait(waitMs);
   289	  const staged = await operations.processDueMessages();
   290	  const reconciled = await reconciliation;
   291	  return {
   292	    claimed: initial.claimed + staged.claimed,
   293	    providerAccepted: initial.providerAccepted + staged.providerAccepted + reconciled.accepted,
   294	    delivered: 0,
   295	    bounced: reconciled.failed,
   296	    stopped: initial.stopped + staged.stopped,
   297	    unresolved: initial.unresolved + staged.unresolved + reconciled.ambiguous,
   298	    regularPrepared: regularPreparation.prepared,
   299	    directPrepared: regularPreparation.directPrepared,
   300	    publicPrepared: regularPreparation.publicPrepared,
   301	    directShortfall: regularPreparation.directShortfall,
   302	    publicShortfall: regularPreparation.publicShortfall,
   303	    regularShortfall: regularPreparation.shortfall,
   304	    hotMarketPrepared: hotMarketPreparation.prepared,
   305	    hotMarketScheduled: hotMarketPreparation.totalScheduled,
   306	    hotMarketShortfall: hotMarketPreparation.shortfall,
   307	    hotLeadTarget: hotLeadPreparation.totalScheduled + hotLeadPreparation.shortfall,
   308	    hotLeadPrepared: hotLeadPreparation.totalScheduled,
   309	    hotLeadShortfall: hotLeadPreparation.shortfall,
   310	    ...(topUp ? { topUpPrepared: topUp.prepared, topUpShortfall: topUp.shortfall } : {}),
   311	    acquisitionErrors,
   312	    verificationPromoted: verificationSummary.promoted,
   313	    verificationFinderCalls: verificationSummary.finderCalls,
   314	    verificationCreditBlocked: verificationSummary.creditBlocked,
   315	    waitMs,
   316	  };
   317	}
