   183	export async function ensureAuthoritativeOutreachConfig(): Promise<
   184	  "disabled" | "present" | "inserted"
   185	> {
   186	  if (process.env.NODE_ENV !== "production") {
   187	    return "disabled";
   188	  }
   189	
   190	  const v1BootstrapEnabled = process.env.OUTREACH_CONFIG_BOOTSTRAP_ENABLED === "true";
   191	  const v2BootstrapEnabled = process.env.OUTREACH_CONFIG_V2_BOOTSTRAP_ENABLED === "true";
   192	  if (!v1BootstrapEnabled && !v2BootstrapEnabled) return "disabled";
   193	
   194	  const [existing] = await db.select({ id: outreachSystemConfigsTable.id })
   195	    .from(outreachSystemConfigsTable)
   196	    .limit(1);
   197	  let changed = false;
   198	  if (!existing && v1BootstrapEnabled) {
   199	    const [inserted] = await db.insert(outreachSystemConfigsTable).values({
   200	      version: 1,
   201	      status: "active",
   202	      policy: AUTHORITATIVE_OUTREACH_POLICY_V1,
   203	    }).onConflictDoNothing().returning({ id: outreachSystemConfigsTable.id });
   204	    changed = Boolean(inserted);
   205	    if (!inserted) {
   206	      const [raced] = await db.select({ id: outreachSystemConfigsTable.id })
   207	        .from(outreachSystemConfigsTable)
   208	        .limit(1);
   209	      if (!raced) throw new Error("Authoritative outreach configuration bootstrap did not create a row");
   210	    }
   211	  }
   212	
   213	  // This is intentionally separate from v1 bootstrap. Owner-approved policy
   214	  // changes update version 2 idempotently while preserving later rows.
   215	  if (process.env.OUTREACH_CONFIG_V2_BOOTSTRAP_ENABLED === "true") {
   216	    const [v2] = await db.select({ id: outreachSystemConfigsTable.id })
   217	      .from(outreachSystemConfigsTable)
   218	      .where(eq(outreachSystemConfigsTable.version, 2))
   219	      .limit(1);
   220	    if (v2) {
   221	      await db.update(outreachSystemConfigsTable).set({
   222	        status: "active",
   223	        policy: AUTHORITATIVE_OUTREACH_POLICY_V2,
   224	      }).where(eq(outreachSystemConfigsTable.id, v2.id));
   225	      changed = true;
   226	    } else {
   227	      const [insertedV2] = await db.insert(outreachSystemConfigsTable).values({
   228	        version: 2,
   229	        status: "active",
   230	        policy: AUTHORITATIVE_OUTREACH_POLICY_V2,
   231	      }).onConflictDoNothing().returning({ id: outreachSystemConfigsTable.id });
   232	      changed = changed || Boolean(insertedV2);
   233	    }
   234	    const [v3] = await db.select({ id: outreachSystemConfigsTable.id })
   235	      .from(outreachSystemConfigsTable)
   236	      .where(eq(outreachSystemConfigsTable.version, 3))
   237	      .limit(1);
   238	    if (!v3) {
   239	      const [insertedV3] = await db.insert(outreachSystemConfigsTable).values({
   240	        version: 3,
   241	        status: "active",
   242	        policy: AUTHORITATIVE_OUTREACH_POLICY_V3,
   243	      }).onConflictDoNothing().returning({ id: outreachSystemConfigsTable.id });
   244	      changed = changed || Boolean(insertedV3);
   245	    }
   246	    const [v4] = await db.select({ id: outreachSystemConfigsTable.id })
   247	      .from(outreachSystemConfigsTable)
   248	      .where(eq(outreachSystemConfigsTable.version, 4))
   249	      .limit(1);
   250	    if (!v4) {
   251	      const [insertedV4] = await db.insert(outreachSystemConfigsTable).values({
   252	        version: 4,
   253	        status: "active",
   254	        policy: AUTHORITATIVE_OUTREACH_POLICY_V4,
   255	      }).onConflictDoNothing().returning({ id: outreachSystemConfigsTable.id });
   256	      changed = changed || Boolean(insertedV4);
   257	    }
   258	  }
   259	  return changed ? "inserted" : "present";
   260	}

   262	export function phoenixMonthKey(now = new Date()): string {
   263	  const parts = Object.fromEntries(new Intl.DateTimeFormat("en-CA", {
   264	    timeZone: "America/Phoenix", year: "numeric", month: "2-digit",
   265	  }).formatToParts(now).map((part) => [part.type, part.value]));
   266	  return `${parts.year}-${parts.month}`;
   267	}
   268	
   269	function resolveSchedule(policy: OutreachPolicy, month: string) {
   270	  const exact = policy.monthlySchedules.find((schedule) => schedule.month === month);
   271	  if (exact) return exact;
   272	  const forward = policy.monthlySchedules
   273	    .filter((schedule) => schedule.month <= month)
   274	    .sort((a, b) => b.month.localeCompare(a.month))[0];
   275	  return forward?.month >= "2027-01" ? forward : undefined;
   276	}

   278	export function phoenixDateKey(now = new Date()): string {
   279	  const parts = Object.fromEntries(new Intl.DateTimeFormat("en-CA", {
   280	    timeZone: "America/Phoenix", year: "numeric", month: "2-digit", day: "2-digit",
   281	  }).formatToParts(now).map((part) => [part.type, part.value]));
   282	  return `${parts.year}-${parts.month}-${parts.day}`;
   283	}

   285	export function effectiveLaneAllocations(
   286	  schedule: OutreachPolicy["monthlySchedules"][number],
   287	  now = new Date(),
   288	): LaneAllocations | undefined {
   289	  if (
   290	    schedule.allocationEffectiveFrom
   291	    && phoenixDateKey(now) < schedule.allocationEffectiveFrom
   292	  ) {
   293	    return schedule.preEffectiveLaneAllocations ?? schedule.laneAllocations;
   294	  }
   295	  return schedule.laneAllocations;
   296	}

   298	export async function loadOutreachSystemConfig(
   299	  now = new Date(),
   300	  executor: Pick<typeof db, "select"> = db,
   301	): Promise<OutreachRuntimeConfig> {
   302	  const [row] = await executor.select().from(outreachSystemConfigsTable)
   303	    .where(eq(outreachSystemConfigsTable.status, "active"))
   304	    .orderBy(desc(outreachSystemConfigsTable.version)).limit(1);
   305	  if (!row) throw new Error("Authoritative outreach configuration is missing; outreach is fail-closed");
   306	  const policy = policySchema.parse(row.policy);
   307	  if (policy.sendingSafeguards.maxMonthly > policy.forwardMonthlyCap) {
   308	    throw new Error("Invalid outreach policy: safeguard exceeds forward monthly cap");
   309	  }
   310	  const month = phoenixMonthKey(now);
   311	  const schedule = resolveSchedule(policy, month);
   312	  if (!schedule) throw new Error(`No authoritative outreach schedule for Phoenix month ${month}`);
   313	  if (schedule.monthlyTarget > policy.forwardMonthlyCap) {
   314	    throw new Error("Invalid outreach policy: monthly target exceeds cap");
   315	  }
   316	  runtime = { version: row.version, policy, month, schedule };
   317	  loadError = null;
   318	  return runtime;
   319	}

   325	export function requiredDailyPace(target: number, sent: number, remainingSendingDays: number): number {
   326	  if (remainingSendingDays <= 0) return Math.max(0, target - sent);
   327	  return Math.max(0, Math.ceil((target - sent) / remainingSendingDays));
   328	}

   330	/** Calendar pacing used by preparation and reservation callers alike. */
   331	export function configuredDailyAllowance(runtimeConfig: OutreachRuntimeConfig, now = new Date()): number {
   332	  const target = Math.min(50_000, runtimeConfig.schedule.monthlyTarget);
   333	  const allocations = effectiveLaneAllocations(runtimeConfig.schedule, now);
   334	  if (allocations?.namedHotMarketShared !== undefined) {
   335	    return allocations.hotLead === OUTREACH_UNCAPPED
   336	      ? allocations.namedHotMarketShared + allocations.public
   337	      : allocations.namedHotMarketShared + allocations.hotLead;
   338	  }
   339	  if (runtimeConfig.month === "2026-09") {
   340	    if (allocations) {
   341	      const namedAndHotMarket = allocations.namedHotMarketShared
   342	        ?? allocations.named + allocations.hotMarket;
   343	      const hotLead = allocations.hotLead === OUTREACH_UNCAPPED ? 0 : allocations.hotLead;
   344	      return namedAndHotMarket + allocations.public + hotLead;
   345	    }
   346	    return 400;
   347	  }
   348	  const parts = new Intl.DateTimeFormat("en-CA", {
   349	    timeZone: "America/Phoenix", year: "numeric", month: "2-digit", day: "2-digit",
   350	  }).formatToParts(now).reduce<Record<string, string>>((result, part) => {
   351	    result[part.type] = part.value;
   352	    return result;
   353	  }, {});
   354	  const year = Number(parts.year);
   355	  const month = Number(parts.month);
   356	  const day = Number(parts.day);
   357	  const daysInMonth = new Date(Date.UTC(year, month, 0)).getUTCDate();
   358	  let remainingDays = 0;
   359	  for (let current = day; current <= daysInMonth; current += 1) {
   360	    const weekday = new Date(Date.UTC(year, month - 1, current)).getUTCDay();
   361	    if (weekday !== 0 && weekday !== 6) remainingDays += 1;
   362	  }
   363	  return Math.min(target, Math.ceil(target / Math.max(1, remainingDays)));
   364	}

   375	/** Engagement is intentionally monotonic: the strongest verified signal wins. */
   376	export function calculateLeadScore(policy: OutreachPolicy, signals: {
   377	  positiveReply?: boolean; reply?: boolean; clickCount?: number; repeatedEngagement?: boolean; openCount?: number; negativeReply?: boolean;
   378	}): number {
   379	  const scoring = policy.scoring;
   380	  if (signals.positiveReply) return scoring.positiveReply ?? scoring.reply ?? 0;
   381	  if (signals.reply) return scoring.reply ?? 0;
   382	  if ((signals.clickCount ?? 0) > 1) return scoring.multipleClicks ?? scoring.click ?? 0;
   383	  if ((signals.clickCount ?? 0) === 1) return scoring.click ?? 0;
   384	  if (signals.repeatedEngagement) return scoring.repeatedEngagement ?? 0;
   385	  if ((signals.openCount ?? 0) > 1) return scoring.multipleOpens ?? 0;
   386	  if ((signals.openCount ?? 0) === 1) return scoring.singleOpen ?? 0;
   387	  return signals.negativeReply ? scoring.negativeReply ?? 0 : 0;
   388	}

   390	export function resolveLeadStatus(signals: {
   391	  suppressed?: boolean; negativeReply?: boolean; positiveReply?: boolean; reply?: boolean; clickCount?: number; openCount?: number;
   392	}): string {
   393	  if (signals.suppressed) return "suppressed";
   394	  if (signals.negativeReply) return "not_interested";
   395	  if (signals.positiveReply) return "qualified";
   396	  if (signals.reply) return "replied";
   397	  if ((signals.clickCount ?? 0) > 0 || (signals.openCount ?? 0) > 0) return "engaged";
   398	  return "cold";
   399	}
