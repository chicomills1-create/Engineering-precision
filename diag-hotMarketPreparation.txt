    39	export function getHotMarketRemainingCapacity(existing: number): number {
    40	  return Math.max(0, HOT_MARKET_DAILY_TARGET - existing);
    41	}
    42	
    43	export function countHotMarketMessages(
    44	  sourceTypes: Array<string | null>,
    45	): number {
    46	  return sourceTypes.filter(isHotMarketSourceType).length;
    47	}
    48	
    49	function canPrepareHotMarket(
    50	  prospect: Prospect,
    51	  campaign: Campaign,
    52	  suppressedEmails: Set<string>,
    53	): boolean {
    54	  const email = prospect.contactEmail?.trim().toLowerCase();
    55	  if (!email || suppressedEmails.has(email)) return false;
    56	  try {
    57	    assertOutreachEligibilityBase({
    58	      id: 0,
    59	      prospectId: prospect.id,
    60	      campaignId: campaign.id,
    61	      sequenceNumber: 1,
    62	      subject: "",
    63	      body: "",
    64	      status: "draft",
    65	      scheduledAt: null,
    66	      sentAt: null,
    67	      providerMessageId: null,
    68	      providerReconciliationKey: null,
    69	      error: null,
    70	      sourceType: HOT_MARKET_SOURCE_TYPE,
    71	      sourceId: null,
    72	      catchUpCohortId: null,
    73	      createdAt: new Date(),
    74	      updatedAt: new Date(),
    75	    }, prospect, campaign, {
    76	      requireApprovedMessage: false,
    77	      requireApprovedProspect: false,
    78	    });
    79	    return true;
    80	  } catch {
    81	    return false;
    82	  }
    83	}

    85	export async function prepareNextPhoenixHotMarketOutreach(
    86	  now = new Date(),
    87	  target?: PhoenixPreparationTarget,
    88	): Promise<{
    89	  state: "skipped" | "completed" | "failed";
    90	  prepared: number;
    91	  totalScheduled: number;
    92	  shortfall: number;
    93	}> {
    94	  if (!target && !isPhoenixPreparationWindowOpen(now)) {
    95	    return { state: "skipped", prepared: 0, totalScheduled: 0, shortfall: 0 };
    96	  }
    97	  const { scheduledAt } = target ?? getNextPhoenixPreparationTarget(now);
    98	  const laneConfig = await getAuthoritativeLaneConfig(undefined, undefined, scheduledAt);
    99	  const targetEnd = new Date(scheduledAt.getTime() + 24 * 60 * 60 * 1000);
   100	  const sharedLimit = getNamedHotMarketSharedLimit(laneConfig);
   101	
   102	  try {
   103	    const rows = await db.select({ prospect: prospectsTable, campaign: campaignsTable })
   104	      .from(prospectsTable)
   105	      .innerJoin(campaignsTable, eq(prospectsTable.campaignId, campaignsTable.id))
   106	      .where(and(
   107	        inArray(prospectsTable.status, ["approved", "review"]),
   108	        eq(prospectsTable.emailStatus, "verified"),
   109	        eq(prospectsTable.contactConfidence, "high"),
   110	        eq(campaignsTable.status, "active"),
   111	        eq(campaignsTable.name, "Verified National Hot-Market Outreach"),
   112	      ));
   113	    const [suppressions, initialMessages, claims, targetInitials] = await Promise.all([
   114	      db.select({ email: outreachSuppressionsTable.email }).from(outreachSuppressionsTable),
   115	      db.select({
   116	        prospectId: outreachMessagesTable.prospectId,
   117	        companyName: prospectsTable.companyName,
   118	        website: prospectsTable.website,
   119	        contactEmail: prospectsTable.contactEmail,
   120	      })
   121	        .from(outreachMessagesTable)
   122	        .innerJoin(prospectsTable, eq(outreachMessagesTable.prospectId, prospectsTable.id))
   123	        .where(eq(outreachMessagesTable.sequenceNumber, 1)),
   124	      db.select({
   125	        prospectId: outreachSequenceSendClaimsTable.prospectId,
   126	        companyName: prospectsTable.companyName,
   127	        website: prospectsTable.website,
   128	        contactEmail: prospectsTable.contactEmail,
   129	      })
   130	        .from(outreachSequenceSendClaimsTable)
   131	        .innerJoin(prospectsTable, eq(outreachSequenceSendClaimsTable.prospectId, prospectsTable.id))
   132	        .where(eq(outreachSequenceSendClaimsTable.sequenceNumber, 1)),
   133	      db.select({
   134	        id: prospectsTable.id,
   135	        contactEmail: prospectsTable.contactEmail,
   136	        companyName: prospectsTable.companyName,
   137	        website: prospectsTable.website,
   138	        state: prospectsTable.state,
   139	        contactName: prospectsTable.contactName,
   140	        contactEvidenceType: prospectsTable.contactEvidenceType,
   141	        fitScore: prospectsTable.fitScore,
   142	        needScore: prospectsTable.needScore,
   143	        sourceType: outreachMessagesTable.sourceType,
   144	      })
   145	        .from(outreachMessagesTable)
   146	        .innerJoin(prospectsTable, eq(outreachMessagesTable.prospectId, prospectsTable.id))
   147	        .where(and(
   148	          eq(outreachMessagesTable.sequenceNumber, 1),
   149	          gte(outreachMessagesTable.scheduledAt, scheduledAt),
   150	          lt(outreachMessagesTable.scheduledAt, targetEnd),
   151	          inArray(outreachMessagesTable.status, ["approved", "sending"]),
   152	        )),
   153	    ]);
   154	    const suppressedEmails = new Set(suppressions.map((row) => row.email.trim().toLowerCase()));
   155	    const [catchUpCohort] = await db.select().from(outreachCatchUpCohortsTable)
   156	      .where(and(
   157	        eq(outreachCatchUpCohortsTable.name, "Apex Grid fresh verified catch-up"),
   158	        eq(outreachCatchUpCohortsTable.status, "active"),
   159	      )).limit(1);
   160	    const usedProspects = new Set([
   161	      ...initialMessages.map((row) => row.prospectId),
   162	      ...claims.map((row) => row.prospectId),
   163	    ]);
   164	    const usedEmails = new Set([
   165	      ...initialMessages.flatMap((row) =>
   166	        row.contactEmail ? [row.contactEmail.trim().toLowerCase()] : []
   167	      ),
   168	      ...claims.flatMap((row) =>
   169	        row.contactEmail ? [row.contactEmail.trim().toLowerCase()] : []
   170	      ),
   171	      ...targetInitials.flatMap((row) =>
   172	      row.contactEmail ? [row.contactEmail.trim().toLowerCase()] : []
   173	      ),
   174	    ]);
   175	    const usedDomains = new Set([
   176	      ...initialMessages.map(companyDomain),
   177	      ...claims.map(companyDomain),
   178	      ...targetInitials.map(companyDomain),
   179	    ]);
   180	    const currentHotMarketCount = countHotMarketMessages(
   181	      targetInitials.map((row) => row.sourceType),
   182	    );
   183	    const sharedNamedHotMarket = laneConfig.namedHotMarketSharedLimit !== undefined;
   184	    const currentNamedCount = sharedNamedHotMarket
   185	      ? targetInitials.filter((row) => getOutreachDailyLane(row, row) === "named").length
   186	      : 0;
   187	    const remainingCapacity = Math.max(0, sharedLimit - currentHotMarketCount - currentNamedCount);
   188	    const eligible = prioritizePreparationCandidates(
   189	      rows
   190	        .filter(({ prospect, campaign }) =>
   191	          isRecurringHotMarketCampaign(campaign)
   192	          && canPrepareHotMarket(prospect, campaign, suppressedEmails)
   193	          && !usedProspects.has(prospect.id))
   194	        .map(({ prospect }) => prospect),
   195	    );
   196	    const campaignByProspect = new Map(rows.map(({ prospect, campaign }) => [prospect.id, campaign]));
   197	    let prepared = 0;
   198	
   199	    for (const prospect of eligible) {
   200	      if (prepared >= remainingCapacity) break;
   201	      const email = prospect.contactEmail!.trim().toLowerCase();
   202	      const domain = companyDomain(prospect);
   203	      if (usedEmails.has(email) || usedDomains.has(domain)) continue;
   204	      const inserted = await db.transaction(async (tx) => {
   205	        await tx.execute(sql`select pg_advisory_xact_lock(hashtextextended(${`hot-market-window:${scheduledAt.toISOString()}`}, 0))`);
   206	        await tx.execute(sql`select pg_advisory_xact_lock(hashtextextended(${email}, 0))`);
   207	        await tx.execute(sql`select pg_advisory_xact_lock(hashtextextended(${domain}, 0))`);
   208	        const currentWindowRows = await tx.select({
   209	          sourceType: outreachMessagesTable.sourceType,
   210	          contactEmail: prospectsTable.contactEmail,
   211	          contactName: prospectsTable.contactName,
   212	          contactEvidenceType: prospectsTable.contactEvidenceType,
   213	        })
   214	          .from(outreachMessagesTable)
   215	          .innerJoin(prospectsTable, eq(outreachMessagesTable.prospectId, prospectsTable.id))
   216	          .where(and(
   217	            eq(outreachMessagesTable.sequenceNumber, 1),
   218	            gte(outreachMessagesTable.scheduledAt, scheduledAt),
   219	            lt(outreachMessagesTable.scheduledAt, targetEnd),
   220	            inArray(outreachMessagesTable.status, ["approved", "sending", "sent", "delivered"]),
   221	          ));
   222	        const currentSharedCount = currentWindowRows.filter((row) => {
   223	          const lane = getOutreachDailyLane(row, row);
   224	          return lane === "hot_market" || (sharedNamedHotMarket && lane === "named");
   225	        }).length;
   226	        if (currentSharedCount >= sharedLimit) return false;
   227	        const [currentProspect] = await tx.select().from(prospectsTable)
   228	          .where(eq(prospectsTable.id, prospect.id))
   229	          .limit(1);
   230	        const campaign = campaignByProspect.get(prospect.id);
   231	        if (!currentProspect || !campaign || !canPrepareHotMarket(
   232	          currentProspect,
   233	          campaign,
   234	          suppressedEmails,
   235	        )) return false;
   236	        const [blocked] = await tx.select({ id: outreachMessagesTable.id })
   237	          .from(outreachMessagesTable)
   238	          .innerJoin(prospectsTable, eq(outreachMessagesTable.prospectId, prospectsTable.id))
   239	          .where(and(
   240	            eq(outreachMessagesTable.sequenceNumber, 1),
   241	            or(
   242	              eq(outreachMessagesTable.prospectId, currentProspect.id),
   243	              sql`lower(trim(${prospectsTable.contactEmail})) = ${email}`,
   244	              sql`lower(${prospectsTable.website}) like ${`%${domain}%`}`,
   245	            ),
   246	          ))
   247	          .limit(1);
   248	        if (blocked) return false;
   249	        const [claimed] = await tx.select({ id: outreachSequenceSendClaimsTable.id })
   250	          .from(outreachSequenceSendClaimsTable)
   251	          .innerJoin(prospectsTable, eq(outreachSequenceSendClaimsTable.prospectId, prospectsTable.id))
   252	          .where(and(
   253	            eq(outreachSequenceSendClaimsTable.sequenceNumber, 1),
   254	            or(
   255	              eq(outreachSequenceSendClaimsTable.prospectId, currentProspect.id),
   256	              sql`lower(trim(${prospectsTable.contactEmail})) = ${email}`,
   257	              sql`lower(${prospectsTable.website}) like ${`%${domain}%`}`,
   258	            ),
   259	          ))
   260	          .limit(1);
   261	        if (claimed) return false;
   262	        const [suppression] = await tx.select({ id: outreachSuppressionsTable.id })
   263	          .from(outreachSuppressionsTable)
   264	          .where(eq(outreachSuppressionsTable.email, email))
   265	          .limit(1);
   266	        if (suppression) return false;
   267	
   268	        if (currentProspect.status === "review") {
   269	          const [promoted] = await tx.update(prospectsTable).set({ status: "approved" })
   270	            .where(and(
   271	              eq(prospectsTable.id, currentProspect.id),
   272	              eq(prospectsTable.status, "review"),
   273	            )).returning({ id: prospectsTable.id });
   274	          if (!promoted) return false;
   275	        }
   276	        let catchUpCohortId: number | null = null;
   277	        if (catchUpCohort) {
   278	          await tx.execute(sql`select ${outreachCatchUpCohortsTable.id}
   279	            from ${outreachCatchUpCohortsTable}
   280	            where ${outreachCatchUpCohortsTable.id} = ${catchUpCohort.id}
   281	            for update`);
   282	          const [reserved] = await tx.select({ count: sql<number>`count(*)::int` })
   283	            .from(outreachCatchUpReservationsTable)
   284	            .where(and(
   285	              eq(outreachCatchUpReservationsTable.cohortId, catchUpCohort.id),
   286	              inArray(outreachCatchUpReservationsTable.status, ["reserved", "accepted"]),
   287	            ));
   288	          if ((reserved?.count ?? 0) < catchUpCohort.targetCount) {
   289	            catchUpCohortId = catchUpCohort.id;
   290	          } else {
   291	            // Active catch-up work is a hard gate: never create an opener
   292	            // outside the cohort once its bounded capacity is exhausted.
   293	            return false;
   294	          }
   295	        }
   296	        const personalization = currentProspect.needSignals!.trim();
   297	        await tx.insert(outreachMessagesTable).values([
   298	          {
   299	            prospectId: currentProspect.id,
   300	            campaignId: campaign.id,
   301	            sequenceNumber: 1,
   302	            subject: hotMarketOutreachSubject(currentProspect.audience, currentProspect.state),
   303	            body: hotMarketOutreachBody({
   304	              contactName: currentProspect.contactName!,
   305	              personalization,
   306	              audience: currentProspect.audience,
   307	            }),
   308	            status: "approved",
   309	            scheduledAt,
   310	            sourceType: HOT_MARKET_RECURRING_SOURCE_TYPE,
   311	            catchUpCohortId,
   312	          },
   313	          ...hotMarketOutreachFollowUps(
   314	            currentProspect.contactName!,
   315	            currentProspect.audience,
   316	            currentProspect.state,
   317	          ).map((followUp) => ({
   318	            prospectId: currentProspect.id,
   319	            campaignId: campaign.id,
   320	            sequenceNumber: followUp.sequenceNumber,
   321	            subject: followUp.subject,
   322	            body: followUp.body,
   323	            status: "approved",
   324	            scheduledAt: null,
   325	            sourceType: HOT_MARKET_RECURRING_SOURCE_TYPE,
   326	          })),
   327	        ]);
   328	        if (catchUpCohortId) {
   329	          const [created] = await tx.select({ id: outreachMessagesTable.id })
   330	            .from(outreachMessagesTable)
   331	            .where(and(
   332	              eq(outreachMessagesTable.prospectId, currentProspect.id),
   333	              eq(outreachMessagesTable.sequenceNumber, 1),
   334	              eq(outreachMessagesTable.catchUpCohortId, catchUpCohortId),
   335	            )).limit(1);
   336	          if (created) await tx.insert(outreachCatchUpReservationsTable).values({
   337	            cohortId: catchUpCohortId,
   338	            messageId: created.id,
   339	          }).onConflictDoNothing();
   340	        }
   341	        return true;
   342	      });
   343	      if (inserted) {
   344	        prepared += 1;
   345	        usedEmails.add(email);
   346	        usedDomains.add(domain);
   347	      }
   348	    }
   349	
   350	    const totalScheduled = currentHotMarketCount + prepared;
   351	    return {
   352	      state: "completed",
   353	      prepared,
   354	      totalScheduled,
   355	      shortfall: Math.max(0, sharedLimit - totalScheduled - currentNamedCount),
   356	    };
   357	  } catch (error) {
   358	    logger.error({ err: error }, "Recurring hot-market preparation failed");
   359	    return {
   360	      state: "failed",
   361	      prepared: 0,
   362	      totalScheduled: 0,
   363	      shortfall: sharedLimit,
   364	    };
   365	  }
   366	}
